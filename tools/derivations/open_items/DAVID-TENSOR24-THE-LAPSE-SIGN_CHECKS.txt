== probe.py ==
"""GR-sign probe: the regulator's time-gear source on light and on dust, integers only.

Reads, never edits, /home/user/Universe24.  Tick.carry is wrapped: before the engine's own carry
runs, the wrapper reads tick.forms, tick.stress, tick.rate, tick.stretch_now (as impulse left them)
and recomputes, per driven train, the forms' parts (u_in u_out and floor(-a F / w)) and, per hand,
the stress trace, checking that they sum exactly to the engine's arrays.  It then calls the engine's
carry unchanged.  Every recorded number is a Python int (sums of the engine's object arrays).

Modes
  head : the engine's run, observed only (the wrapper changes nothing).
  test : a probe-only counterfactual, the "test field": the regulator's own act is skipped (Clock.act
         on the 7-gear regulator returns its gears unchanged) and its gears are put back to the
         vacuum after every tick, so X = Gamma, p0 = Gamma at every clock and tick, no turnover
         fires, and the hands run Rule3 freely on the vacuum's lattice.  The source is still read
         exactly as the engine reads it (same forms, same stress, same held_numerator).
"""

def isum(a) -> int:
    """Exact integer sum of an object array (or an int)."""
    if np.ndim(a) == 0:
        return int(a)
    return int(sum(int(x) for x in np.asarray(a).reshape(-1)))


def train_name(clock: Clock, r) -> str:
    idx = clock.by_index.index(r) if r in clock.by_index else None
    if idx is None and r.twin is not None:
        idx = clock.by_index.index(r.twin)
    return f"row{idx}"


def probe_carry(self) -> None:
    """Read the tick's source as impulse left it, then run the engine's carry unchanged."""
    clk = self.clock
    K, rim, G = clk.trace, clk.rim, clk.full_rate
    k_reg = self.where[id(self.regulator)]
    X, rate = self.stretch_now, self.rate
    trace, _traceless = tensor_parts(self.stress[:AXES])

    # --- per-train breakdown, recomputed with the impulse's own formulas -------------------
    d = (clk.teeth + clk.teeth) * X
    w_side = readings.mean_teeth(clk, self.regulator.gears[0])
    forms_sum = zeros(clk.size)
    trace_sum = zeros(clk.size)
    per = {}
    for r in self.driven:
        k = self.where[id(r)]
        p_side = [w_side * (g.now - g.before) + g.fine - mid(w_side) for g in self.trains[k]]
        u_in = [rates.rounded(p + p + 1, d) for p in p_side]
        u_out = [rates.rounded(p + p + 1, d) for p in self.impelled[k]]
        kin = zeros(clk.size)
        pot = zeros(clk.size)
        for prod, pull in zip(
            hands.products(u_in, u_out), hands.products(self.starts[k], self.pull[k]), strict=True
        ):
            kin = kin + prod
            pot = pot + carries(-pull, self.train_teeth[k])
        forms_sum = forms_sum + kin + pot
        entry = {"kin": isum(kin), "pot": isum(pot)}
        if r in self.hands:
            velocity = [x + y for x, y in zip(u_in, u_out, strict=True)]
            st = stress_of(self.starts[k], velocity)
            tr = r.ratio[0] * sum(st[:AXES])
            trace_sum = trace_sum + tr
            entry["trace"] = isum(tr)
            entry["P"] = isum(rate * rate * tr)
        per[train_name(clk, r)] = entry
    exact_forms = bool(np.all(forms_sum == self.forms))
    exact_trace = bool(np.all(trace_sum == trace))

    # --- the source as the engine forms it ------------------------------------------------
    E = K * rim * G * G * self.forms  # the forms' term, K rim Gamma^2 Phi
    P = rate * rate * trace  # the stress term, p0^2 tau
    B_head, B_flip = E - P, E + P
    N_head = held_numerator(K, X, self.forms, trace, rate, rim, G)  # the engine's own numerator
    exact_numerator = bool(np.all(N_head == HELD_DRIVE_WEIGHT * X * rate * B_head))
    N_flip = HELD_DRIVE_WEIGHT * X * rate * B_flip
    hold = X * self.trains[k_reg][0].dteeth
    clamp_head = (N_head > hold) | (N_head < -hold)
    clamp_flip = (N_flip > hold) | (N_flip < -hold)
    absN = np.abs(N_head)
    nz = self.forms != 0
    rec = {
        "tick": clk.ticks,
        "Phi": isum(self.forms),
        "tau": isum(trace),
        "E": isum(E),
        "P": isum(P),
        "B_head": isum(B_head),
        "B_flip": isum(B_flip),
        "N_head": isum(N_head),
        "N_flip": isum(N_flip),
        "clamp_head": int(clamp_head.sum()),
        "clamp_flip": int(clamp_flip.sum()),
        # the largest |N| / hold at any clock, as an exact fraction's float (head and flipped)
        "max_N_over_hold_head": float(max(Fraction(int(a), int(h)) for a, h in zip(absN.flat, hold.flat))),
        "max_N_over_hold_flip": float(
            max(Fraction(abs(int(a)), int(h)) for a, h in zip(N_flip.flat, hold.flat))
        ),
        "X_min": int(min(X.flat)),
        "X_max": int(max(X.flat)),
        "p0_min": int(min(np.asarray(rate).flat)),
        "p0_max": int(max(np.asarray(rate).flat)),
        "forms_clocks": int(nz.sum()),
        "per": per,
        "exact": exact_forms and exact_trace and exact_numerator,
    }
    STATE["records"].append(rec)
    STATE["forms_mask"] = nz.copy()
    STATE["forms_arr"] = self.forms.copy()
    _orig_carry(self)


def probe_act(self, gears, r, meshes, numerators, going, flow=None, top=None, kept=None):
    """Test mode only: the regulator's act skipped (its gears returned unchanged)."""
    if STATE["mode"] == "test" and len(gears) == REGULATOR:
        same = [g.copy() for g in gears]
        return gears, same
    return _orig_act(self, gears, r, meshes, numerators, going, flow, top, kept)

def build(world: str, shape, seed, universe: str | None = None) -> Clock:
    calibre, piece = world_of(world)
    if universe is not None:  # the piece's keys kept, the calibre read from another universe file
        calibre = json.loads((Path("/home/user/Universe24/tests/worlds") / universe).read_text())
    return Clock(calibre, {**piece, "shape": list(shape), "seed": seed})


def vacuum_regulator(clock: Clock) -> list[Gear]:
    return [g.copy() for g in clock.train(REGULATOR).gears]


def run(
    name: str, world: str, shape, seed, ticks: int, mode: str, universe: str | None = None, setter=None
) -> dict:
    STATE["mode"], STATE["records"] = mode, []
    clock = build(world, shape, seed, universe)
    if setter is not None:  # a probe-made setting laid by the engine's own advance act
        setter(clock)
    vac = vacuum_regulator(clock)
    lay_log = [e for e in clock.log if e.get("event") == "lay"]
    t0 = time.time()
    for _ in range(ticks):
        clock.tick()
        rec = STATE["records"][-1]
        turned = clock.turned_over
        mask = STATE["forms_mask"]
        rec["turned"] = int(turned.sum())
        rec["turned_where_forms"] = int((turned & mask).sum())
        rec["Phi_at_turned"] = isum(np.where(turned, STATE["forms_arr"], 0))
        rec["births"] = clock.counts["births"]
        rec["laid"] = clock.counts["laid"]
        if mode == "test":  # the regulator back to the vacuum, every gear and tooth
            for g, v in zip(clock.train(REGULATOR).gears, vac, strict=True):

def summary(out: dict, skip: int = 0, blocks: int = 5) -> dict:
    """Time averages as exact fractions of the integer sums; the spread over blocks and ticks."""
    recs = out["records"][skip:]
    E = sum(r["E"] for r in recs)
    Bh = sum(r["B_head"] for r in recs)
    Bf = sum(r["B_flip"] for r in recs)
    P = sum(r["P"] for r in recs)
    head = Fraction(Bh, E)
    flip = Fraction(Bf, E)
    x = Fraction(P, E)
    per_tick = [Fraction(r["B_head"], r["E"]) for r in recs if r["E"]]
    n = len(recs) // blocks
    block = []
    for b in range(blocks):
        part = recs[b * n : (b + 1) * n]
        eb = sum(r["E"] for r in part)
        block.append(float(Fraction(sum(r["B_head"] for r in part), eb)))
    kin = sum(sum(v["kin"] for v in r["per"].values()) for r in recs)
    pot = sum(sum(v["pot"] for v in r["per"].values()) for r in recs)
    pt = [float(v) for v in per_tick]
    return {
        "name": out["name"],
        "mode": out["mode"],
        "ticks_used": len(recs),
        "head": float(head),
        "flip": float(flip),
        "x": float(x),
        "head_frac": f"{head.numerator}/{head.denominator}" if head.denominator < 10**12 else None,
        "block_head": block,
        "block_spread": max(block) - min(block),
        "tick_head_mean": float(np.mean(pt)),
        "tick_head_std": float(np.std(pt)),
        "tick_head_min": min(pt),
        "tick_head_max": max(pt),
        "kin_over_pot": float(Fraction(kin, pot)) if pot else None,
        "clamp_head_total": sum(r["clamp_head"] for r in recs),
        "clamp_flip_total": sum(r["clamp_flip"] for r in recs),
        "max_N_over_hold_head": max(r["max_N_over_hold_head"] for r in recs),
        "max_N_over_hold_flip": max(r["max_N_over_hold_flip"] for r in recs),

        "turned_where_forms_mean": float(np.mean([r["turned_where_forms"] for r in recs])),

L^3 boards, laid with the engine's own integer angle pairs (setting.rim_point, meshes.offset_pair,

def plane_setter(row, n_vec, amplitude, sense=1):
    """A travelling plane wave z = A e^{i(k.x - omega t)} on the hand of the row, k = 2 pi n / L
    (sense 1); at sense 0 the conjugate pair (both senses summed, the twelve gears cancelling: a real
    travelling wave on the three gear, no sweep and no lead, as the head's rest lay sets a sprung hand)."""
    def setter(clock):
        L, rim = clock.cube, clock.rim
        train = clock.by_index[row]
        rec = clock.hands().index(train)
        points, along = [], []
        for n in n_vec:
            h = int(round(rim * math.sin(math.pi * n / L)))   # h = rim sin(k/2): the stride's input integer
            pt = rim_point(rim, h)
            points.append(pt)
            seq = [(rim, 0)]
            for _ in range(1, L):
                c, s = offset_pair(seq[-1][0], seq[-1][1], pt[0], pt[1], rim)
                seq.append((int(c), int(s)))
            along.append(seq)
        rot = running_rotation(train.ratio[0], rim, points)
        C, S = zeros(clock.size), zeros(clock.size)
        for at in np.ndindex(*clock.size):
            C[at], S[at] = angle_of(rim, [along[a][at[a]] for a in range(3)])
        Cb, Sb = offset_pair(C, S, rot[0], rot[1], rim)
        base = zeros(clock.size) + amplitude
        for sn in ((1,) if sense else (-1, 1)):
            gears = hands.moving(base, (C, S), (Cb, Sb), rim, sn)
            advance(clock, rec, Ellipsis, tuple(x for g in gears for x in (g.now, g.before)))
    return setter

def predicted(rho, K, L, n_vec):
    k = [2 * math.pi * n / L for n in n_vec]
    cw = rho * sum(math.cos(q) for q in k) / 3
    s2 = 1 - cw * cw
    return rho * 2 * sum(math.sin(q) ** 2 for q in k) / (K * s2)

"""The lattice per-mode prediction for each packet's own mode content (floats: a prediction only, the
engine's numbers stay integer): the laid hand's gears FFT'd at the start (now, and before corrected by
its fine tooth: velocity = P / w_side), each mode k of each gear a free Rule3 oscillator with
cos omega_k = rho_h mean_a cos k_a; its conserved forms Phi_k = |a_n|^2 + |a_{n-1}|^2 - 2 cos w Re(a_n a*_{n-1})
(the engine's u_in u_out + floor(-aF/w) in Fourier form) and its time-mean stress rho_h 4 SUM_a sin^2 k_a
<|a_k|^2>; x_pred = SUM stress / (K SUM Phi).  Compared with the test field's measured time-mean x."""
import json, sys
import numpy as np
sys.path.insert(0, str(__import__("pathlib").Path(__file__).resolve().parent))

             "light_x_q100", "light_diag_q100", "dust_row2_q100", "dust_row3_q100"]:
    world, shape, seed = run_all.SETTINGS[name]
    clock = probe.build(world, shape, seed)
    row = seed[0]["row"]
    train = clock.by_index[row]
    rho = train.ratio[0] / train.ratio[1]
    K = clock.trace
    w_side = int(readings.mean_teeth(clock, clock.train(REGULATOR).gears[0]).flat[0])
    L = clock.cube
    k1 = 2 * np.pi * np.fft.fftfreq(L) # per-axis wave numbers
    kx, ky, kz = np.meshgrid(k1, k1, k1, indexing="ij")
    cosw = rho * (np.cos(kx) + np.cos(ky) + np.cos(kz)) / 3
    sinw = np.sqrt(np.maximum(1 - cosw ** 2, 0))
    s2 = np.sin(kx) ** 2 + np.sin(ky) ** 2 + np.sin(kz) ** 2
    phi = 0.0; stress = 0.0; phi_alt = 0.0
    for g in train.gears:
        now = np.array(g.now, dtype=float)
        vel = (w_side * (np.array(g.now, dtype=object) - np.array(g.before, dtype=object))
               + np.array(g.fine, dtype=object) - mid(w_side)).astype(float) / w_side
        a_n, a_b = np.fft.fftn(now), np.fft.fftn(now - vel)
        Phi_k = np.abs(a_n) ** 2 + np.abs(a_b) ** 2 - 2 * cosw * np.real(a_n * np.conj(a_b))
        ok = sinw > 1e-12
        A = np.where(ok, (a_n * np.exp(1j * np.arccos(np.clip(cosw, -1, 1))) - a_b) / (2j * np.where(ok, sinw, 1)), 0)
        B = a_n - A
        mean_sq = np.where(ok, np.abs(A) ** 2 + np.abs(B) ** 2, np.abs(a_n) ** 2)
        phi += Phi_k.sum(); stress += (rho * 4 * s2 * mean_sq).sum()
        phi_alt += np.where(ok, 2 * (np.abs(A) ** 2 + np.abs(B) ** 2) * sinw ** 2, Phi_k).sum()
    x_pred = stress / (K * phi)
    meas = summ[f"{name}_test"]["x"]
    rows.append({"name": name, "rho_h": rho, "x_pred_modes": x_pred, "x_test_measured": meas,

    print(json.dumps(rows[-1]))
