def intertwiners(k1, k2, k):

def scaled_to_integers(c):
    """The intertwiner times the least common denominator of its entries, and divided by the entries' common factor: an
    integer (Gaussian) matrix with no number chosen."""
    from math import gcd

    den = 1
    for r in c:
        for x in r:
            for part in (x.re, x.im):
                den = den * part.denominator // gcd(den, part.denominator)
    out = [[x * den for x in r] for r in c]
    common = 0
    for r in out:
        for x in r:
            common = gcd(common, int(x.re))
            common = gcd(common, int(x.im))
    return [[C(x.re / common, x.im / common) for x in r] for r in out] if common else out

def inverse(m):
    """The exact inverse of a small complex-rational matrix, by elimination."""
    n = len(m)
    a = [list(r) + [ONE if i == j else ZERO for j in range(n)] for i, r in enumerate(m)]
    for col in range(n):
        piv = next(i for i in range(col, n) if not a[i][col].zero())
        a[col], a[piv] = a[piv], a[col]
        inv = ONE / a[col][col]
        a[col] = [x * inv for x in a[col]]
        for i in range(n):
            if i != col and not a[i][col].zero():
                fac = a[i][col]
                a[i] = [x - fac * y for x, y in zip(a[i], a[col])]
    return [r[n:] for r in a]

def null_space(rows, n):
    """The null space of a linear system over the complex rationals, by exact elimination; returns a basis."""
    a = [list(r) for r in rows]
    pivots = []
    r = 0
    for c in range(n):
        p = next((i for i in range(r, len(a)) if not a[i][c].zero()), None)
        if p is None:
            continue
        a[r], a[p] = a[p], a[r]
        inv = ONE / a[r][c]
        a[r] = [x * inv for x in a[r]]
        for i in range(len(a)):
            if i != r and not a[i][c].zero():
                f = a[i][c]
                a[i] = [x - f * y for x, y in zip(a[i], a[r])]
        pivots.append(c)
        r += 1
    free = [c for c in range(n) if c not in pivots]
    basis = []
    for fc in free:
        v = [ZERO] * n
        v[fc] = ONE
        for i, pc in enumerate(pivots):
            v[pc] = -a[i][fc]
        basis.append(v)
    return basis


def invariant_forms(gens, n, parity=None):
    """Every bilinear form B on the kind with M^T B M = s B for every generator (M, s); parity +1 or -1 adds B^T = parity B."""
    rows = []
    for m, s in gens:
        # (M^T B M)_ij - s B_ij = sum_kl M_ki B_kl M_lj - s B_ij, linear in the n*n unknowns B_kl
        for i in range(n):
            for j in range(n):
                row = [ZERO] * (n * n)
                for k in range(n):
                    for l in range(n):
