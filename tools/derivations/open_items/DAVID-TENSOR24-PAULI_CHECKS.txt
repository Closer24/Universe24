class C:
    """An exact complex rational; with integer parts it is a Gaussian integer, a hand's two gears (three, twelve)."""

    __slots__ = ("re", "im")

    def __init__(self, re=0, im=0):
        self.re, self.im = F(re), F(im)

    def __add__(self, o):
        o = cc(o)
        return C(self.re + o.re, self.im + o.im)

    __radd__ = __add__

    def __sub__(self, o):
        o = cc(o)
        return C(self.re - o.re, self.im - o.im)

    def __rsub__(self, o):
        return cc(o) - self

    def __neg__(self):
        return C(-self.re, -self.im)

    def __mul__(self, o):
        o = cc(o)
        return C(self.re * o.re - self.im * o.im, self.re * o.im + self.im * o.re)

    __rmul__ = __mul__

    def __truediv__(self, o):
        o = cc(o)
        n = o.re * o.re + o.im * o.im
        return self * C(o.re / n, -o.im / n)

    def conj(self):
        return C(self.re, -self.im)

    def norm(self):
        return self.re * self.re + self.im * self.im

    def zero(self):
        return self.re == 0 and self.im == 0

    def __eq__(self, o):
        o = cc(o)
        return self.re == o.re and self.im == o.im

    def __hash__(self):
        return hash((self.re, self.im))

    def __repr__(self):
        return f"({self.re}{'+' if self.im >= 0 else ''}{self.im}i)"

def null_space(rows, n):
    """The null space of a linear system over the complex rationals, by exact elimination; returns a basis."""
    a = [list(r) for r in rows]
    pivots = []
    r = 0
    for c in range(n):
        p = next((i for i in range(r, len(a)) if not a[i][c].zero()), None)
        if p is None:
            continue
        a[r], a[p] = a[p], a[r]
        inv = ONE / a[r][c]
        a[r] = [x * inv for x in a[r]]
        for i in range(len(a)):
            if i != r and not a[i][c].zero():
                f = a[i][c]
                a[i] = [x - f * y for x, y in zip(a[i], a[r])]
        pivots.append(c)
        r += 1
    free = [c for c in range(n) if c not in pivots]
    basis = []
    for fc in free:
        v = [ZERO] * n
        v[fc] = ONE
        for i, pc in enumerate(pivots):
            v[pc] = -a[i][fc]
        basis.append(v)
    return basis


def invariant_forms(gens, n, parity=None):
    """Every bilinear form B on the kind with M^T B M = s B for every generator (M, s); parity +1 or -1 adds B^T = parity B."""
    rows = []
    for m, s in gens:
        # (M^T B M)_ij - s B_ij = sum_kl M_ki B_kl M_lj - s B_ij, linear in the n*n unknowns B_kl
        for i in range(n):
            for j in range(n):
                row = [ZERO] * (n * n)
                for k in range(n):
                    for l in range(n):
                        row[k * n + l] = row[k * n + l] + m[k][i] * m[l][j]
                row[i * n + j] = row[i * n + j] - cc(s)
                rows.append(row)
    if parity is not None:
        for i in range(n):
            for j in range(n):
                row = [ZERO] * (n * n)
                row[i * n + j] = row[i * n + j] + ONE
                row[j * n + i] = row[j * n + i] - cc(parity)
                rows.append(row)
    return null_space(rows, n * n)

"""Exact arithmetic for an independent probe of the Pauli derivation.

Gaussian numbers are (re, im) pairs whose parts are Python ints (Gaussian integers) or
fractions.Fraction (the field Q(i), needed only for elimination, shares and Gram-Schmidt).
A float, a complex or any other type is refused at construction, so no inexact number can
enter. Matrices are lists of rows of G. Written from the definitions given to the second
hand; it shares no code with any other probe.
"""

from __future__ import annotations

from fractions import Fraction


def _exact(x):
    t = type(x)
    if t is int or t is Fraction:
        return x
    raise TypeError(f"inexact number {x!r} of type {t.__name__} refused")


class G(tuple):
    """re + i im as the pair (re, im), parts int or Fraction only."""

    __slots__ = ()

    def __new__(cls, re=0, im=0):
        return tuple.__new__(cls, (_exact(re), _exact(im)))

    @property
    def re(self):
        return self[0]

    @property
    def im(self):
        return self[1]

    def __add__(self, o):
        o = lift(o)
        return G(self[0] + o[0], self[1] + o[1])

    def __radd__(self, o):
        return self.__add__(o)

    def __sub__(self, o):
        o = lift(o)
        return G(self[0] - o[0], self[1] - o[1])

    def __rsub__(self, o):
        o = lift(o)
        return G(o[0] - self[0], o[1] - self[1])

    def __neg__(self):
        return G(-self[0], -self[1])

    def __mul__(self, o):
        o = lift(o)
        a, b = self[0], self[1]
        c, d = o[0], o[1]
        return G(a * c - b * d, a * d + b * c)

    def __rmul__(self, o):
        return self.__mul__(o)

    def conj(self):
        return G(self[0], -self[1])

    def norm(self):
        return self[0] * self[0] + self[1] * self[1]

    def inv(self):
        n = self.norm()
        if n == 0:
            raise ZeroDivisionError("inverse of 0")
        return G(Fraction(self[0]) / n, Fraction(-self[1]) / n)

    def __truediv__(self, o):
        return self * lift(o).inv()

    def zero(self):
        return self[0] == 0 and self[1] == 0

    def real(self):
        return self[1] == 0

    def __repr__(self):
        return f"G({self[0]}, {self[1]})"


def lift(x):
    return x if isinstance(x, G) else G(x, 0)


I = G(0, 1)
ONE = G(1, 0)
ZERO = G(0, 0)


# ---------------------------------------------------------------- matrices over Q(i)
def mat(rows):
    return [[lift(x) for x in r] for r in rows]


def eye(n):
    return [[ONE if r == c else ZERO for c in range(n)] for r in range(n)]


def mmul(A, B):
    return [[sum((A[r][k] * B[k][c] for k in range(len(B))), ZERO) for c in range(len(B[0]))] for r in range(len(A))]


def tr(A):
    return sum((A[k][k] for k in range(len(A))), ZERO)


def transpose(A):
    return [list(r) for r in zip(*A)]


def conjm(A):
    return [[x.conj() for x in r] for r in A]


def dagger(A):
    return transpose(conjm(A))


def scal(c, A):
    c = lift(c)
    return [[c * x for x in r] for r in A]


def madd(A, B):
    return [[x + y for x, y in zip(ra, rb)] for ra, rb in zip(A, B)]


def msub(A, B):
    return [[x - y for x, y in zip(ra, rb)] for ra, rb in zip(A, B)]


def meq(A, B):
    return all(x == y for ra, rb in zip(A, B) for x, y in zip(ra, rb)) and len(A) == len(B)


def kron(A, B):
    """(A kron B)[(i,k),(j,l)] = A[i][j] B[k][l]."""
    nb, mb = len(B), len(B[0])
    return [[A[r // nb][c // mb] * B[r % nb][c % mb] for c in range(len(A[0]) * mb)] for r in range(len(A) * nb)]


def blockdiag(A, B):
    n, m = len(A), len(B)
    out = [[ZERO] * (n + m) for _ in range(n + m)]
    for r in range(n):
        for c in range(n):
            out[r][c] = A[r][c]
    for r in range(m):
        for c in range(m):
            out[n + r][n + c] = B[r][c]
    return out


def det(A):
    """Exact determinant over Q(i) by elimination."""
    A = [list(r) for r in A]
    n, d = len(A), ONE
    for c in range(n):
        p = next((r for r in range(c, n) if not A[r][c].zero()), None)
        if p is None:
            return ZERO
        if p != c:
            A[c], A[p] = A[p], A[c]
            d = -d
        d = d * A[c][c]
        inv = A[c][c].inv()
        for r in range(c + 1, n):
            if not A[r][c].zero():
                f = A[r][c] * inv
                A[r] = [x - f * y for x, y in zip(A[r], A[c])]
    return d


def nullspace(rows, ncols):
    """A basis of {v : rows v = 0} over Q(i), by exact reduced row echelon form."""
    A = [list(r) for r in rows]
    pivots, r = [], 0
    for c in range(ncols):
        p = next((k for k in range(r, len(A)) if not A[k][c].zero()), None)
        if p is None:
            continue
        A[r], A[p] = A[p], A[r]
        inv = A[r][c].inv()
        A[r] = [x * inv for x in A[r]]
        for k in range(len(A)):
            if k != r and not A[k][c].zero():
                f = A[k][c]
                A[k] = [x - f * y for x, y in zip(A[k], A[r])]
        pivots.append(c)
        r += 1
        if r == len(A):
            break
    basis = []
    for f in (c for c in range(ncols) if c not in pivots):
        v = [ZERO] * ncols
        v[f] = ONE
        for k, pc in enumerate(pivots):
            v[pc] = -A[k][f]
        basis.append(v)
    return basis


# ---------------------------------------------------------------- spinors, readings
def inner(u, v):
    """<u, v> = sum of conj(u_a) v_a."""
    return sum((a.conj() * b for a, b in zip(u, v)), ZERO)


def bil(u, B, v):
    """The bilinear form u^T B v (no conjugation)."""
    return sum((u[p] * B[p][q] * v[q] for p in range(len(u)) for q in range(len(v))), ZERO)


def nrm(u):
    """N(u) = sum of N(u_a)."""
    return sum((a.norm() for a in u), 0)


def T(phi):
    """The twins' phase reversal T(z1, z2) = (-conj z2, conj z1)."""
    z1, z2 = phi
    return (-z2.conj(), z1.conj())


def board(n, b, t=None):
    """The board's phase reversal of a pair's gears: (n, b) -> (T(b), T(n)); T looked up at call time."""
    t = T if t is None else t
    return t(b), t(n)


def Dmatrix(n, b):
    """D_ac = i (conj(b_a) n_c - conj(n_a) b_c)."""
    return [[I * (b[a].conj() * n[c] - n[a].conj() * b[c]) for c in (0, 1)] for a in (0, 1)]


class Reading:
    __slots__ = ("D", "W", "S", "det", "hermitian")

    def __init__(self, n, b):
        D = Dmatrix(n, b)
        self.D = D
        self.hermitian = D[0][0].real() and D[1][1].real() and D[1][0] == D[0][1].conj()
        self.W = D[0][0].re + D[1][1].re
        self.S = (2 * D[0][1].re, -2 * D[0][1].im, D[0][0].re - D[1][1].re)
        d = D[0][0] * D[1][1] - D[0][1] * D[1][0]
        self.hermitian = self.hermitian and d.real()
        self.det = d.re

    def ws(self):
        return (self.W, self.S)

    def zero(self):
        return all(x.zero() for r in self.D for x in r)


def dot3(a, b):
    return sum((x * y for x, y in zip(a, b)), 0)


def split_share(W, S, Wd, m):
    """(W W_d + S . m) / (2 W W_d), exact."""
    return Fraction(W * Wd + dot3(S, m), 2 * W * Wd)


class Tally:
    """A claim's count: cases and failures, or a mutant's count: cases and catches."""

    rows: list = []

    def __init__(self, label, mutant=False):
        self.label, self.mutant, self.cases, self.bad = label, mutant, 0, 0
        Tally.rows.append(self)

    def add(self, ok):
        self.cases += 1
        if not ok:
            self.bad += 1
        return ok

    def line(self):
        if self.mutant:
            return f"  MUTANT {self.label}: caught {self.bad} of {self.cases}" + ("" if self.bad else "   <-- NOT CAUGHT")
        return f"  {self.label}: cases {self.cases}, failures {self.bad}" + ("" if not self.bad else "   <-- FAILS")

    @classmethod
    def verdict(cls):
        claims_bad = [t for t in cls.rows if not t.mutant and t.bad]
        mutants_missed = [t for t in cls.rows if t.mutant and not t.bad]
        empty = [t for t in cls.rows if t.cases == 0]
        return claims_bad, mutants_missed, empty

L = 5


def gram_schmidt(vs):
    out = []
    for a in vs:
        v = list(a)
        for w in out:
            c = inner(w, v) / nrm(w)
            v = [x - c * y for x, y in zip(v, w)]
        if all(x.zero() for x in v):
            return None
        out.append(tuple(v))
    return out


def occupation(hs, local=False):
    """O(x) = sum_k phi_k(x) phi_k(x)^dagger / N(phi_k), as (O_uu, O_dd, O_ud) per node-clock."""
    out = []
    for x in range(L):
        ouu = odd = Fraction(0)
        oud = G(Fraction(0), Fraction(0))
        for h in hs:
            u, d = h[2 * x], h[2 * x + 1]
            nk = (u.norm() + d.norm()) if local else nrm(h)
            if nk == 0:
                continue
            ouu += Fraction(u.norm()) / nk
            odd += Fraction(d.norm()) / nk
            oud = oud + (u * d.conj()) / nk
        out.append((ouu, odd, oud))
    return out


def le_one(o):
    ouu, odd, oud = o
    return (2 - ouu - odd) >= 0 and (1 - ouu) * (1 - odd) - oud.norm() >= 0

def T_arr(zu, zd, minus=True):
    """T on a spinor of Gaussian arrays z = (re, im): (-conj z_d, conj z_u); minus=False is the mutant (conj z_d, conj z_u)."""
    s = -1 if minus else 1
    return (s * zd[0], -s * zd[1]), (zu[0], -zu[1])


def hand_arrays(gu, gd):
    nu, nd = (gu[0].now, gu[1].now), (gd[0].now, gd[1].now)
    bu, bd = (gu[0].before, gu[1].before), (gd[0].before, gd[1].before)
    return nu, nd, bu, bd


def to_gears(nu, nd, bu, bd):
    up = [Gear(nu[0], bu[0], None, None), Gear(nu[1], bu[1], None, None)]
    down = [Gear(nd[0], bd[0], None, None), Gear(nd[1], bd[1], None, None)]
    return up, down


def board_arr(gu, gd, minus=True, exchange=True):
    """The board's phase reversal on the real gear arrays: now' = T(before), before' = T(now)."""
    nu, nd, bu, bd = hand_arrays(gu, gd)
    tnu, tnd = T_arr(nu, nd, minus)
    tbu, tbd = T_arr(bu, bd, minus)
    if exchange:

    return to_gears(tnu, tnd, tbu, tbd)
