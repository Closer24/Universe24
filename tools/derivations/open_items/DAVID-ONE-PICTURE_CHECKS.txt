"""The regulator's sources as one tensor act: T(mu, nu) = SUM over records of pinion Re(conj(d_mu z) d_nu z), the centred gradients' one bilinear per index pair, in place of stress_parts and shear_parts; each committed chart regenerated under it against the committed chart (argv: the chart)."""
import json, os, sys, time
sys.set_int_max_str_digits(0)
WT = os.environ.get("WT", "/tmp/claude-0/-home-user-Universe24/2ac4c1db-b416-55d7-b373-04f98f505363/scratchpad/wt_g4")

from event_universe import regulation, hands
from event_universe.core.spokes import AXES, transverse
from event_universe.meshes import gradient
from event_universe.trains import zeros
import event_universe.tick as tick


def tensor(rows, shape, pairs):
    """One act: for each index pair (a, b), SUM over records of pinion Re(conj(d_a z) d_b z)."""
    found = [zeros(shape) for _ in pairs]
    for levels, num in rows:
        d = [[gradient(x, a) for x in levels] for a in range(AXES)]
        for k, (a, b) in enumerate(pairs):
            found[k] = found[k] + num * hands.dot(d[a], d[b])
    return found


def stress_parts(rows, shape):
    return tensor(rows, shape, [(a, a) for a in range(AXES)])


def shear_parts(rows, shape):
    return tensor(rows, shape, [(a, transverse(a)[0]) for a in range(AXES)])


regulation.stress_parts, regulation.shear_parts = stress_parts, shear_parts
tick.stress_parts, tick.shear_parts = stress_parts, shear_parts
