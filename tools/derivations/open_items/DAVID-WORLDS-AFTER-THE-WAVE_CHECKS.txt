def merger_carried():
    """Item 2: the unspringing carries its excess in EX-S1's one remainder, signed: at each end of a merger the clock's remainder gains the sprung take's fall less the light's rise there (fall - wall less rise - wall), logged on the birth as `slack_carried`; the undo made additive in the remainder (every take's restore the remainder less its own fall - booked) and each carried excess taken back at its own entry, so the order is free."""
    import event_universe.striking as sk
    from event_universe import hands
    from event_universe.unwind import advance, move, recoil
    from event_universe.unspring import fitted_notches
    real_fitted = sk.fitted

    def fitted(node, at, kids, take, rec, c, factors):
        before = len(node.log)
        touched = real_fitted(node, at, kids, take, rec, c, factors)
        for entry in node.log[before:]:
            if entry.get("event") == "birth" and "takes" in entry and "exact_lay" in entry:
                carried = []
                for t in entry["takes"]:
                    end = tuple(t["node"])
                    lay = entry["exact_lay"][0] if list(end) == list(entry["minus"]) else entry["exact_lay"][1]
                    if lay is None:
                        continue
                    delta = int(t["taken"]) - int(lay[2])
                    node.slack[end] += delta
                    carried.append([list(end), delta])
                entry["slack_carried"] = carried
        return touched

    def undo(node, frame):
        rec0 = node.hands().index(node.unsprung())
        unbooked = set()
        for entry in reversed(node.log[frame["log_from"]:]):
            at = tuple(entry["node"])
            for end, delta in entry.get("slack_carried", []):
                node.slack[tuple(end)] -= delta
            if entry["event"] == "birth":
                s, (a, b) = entry["line_pair"], entry["amplitude"]
                for where, dp, clock in zip((entry["plus"], entry["minus"]), entry["dP_lay_ends"][::-1], entry["clock_ends"][::-1], strict=True):
                    recoil(node, tuple(where), dp, clock, -1)
                exact = entry.get("exact_lay", [None, None])
                for where, sense, laid in zip((entry["minus"], entry["plus"]), (-1, 1), exact, strict=True):
                    if laid is None:
                        p = s if sense < 0 else hands.twin_of(node.hands(), s)
                        advance(node, p, tuple(where), fitted_notches(node.hands()[p], sense, a, b), -1)
                    else:
                        move(node, s, laid[0], tuple(where), -laid[1])
                if "cross" in entry:
                    cross = entry["cross"]
                    recoil(node, tuple(cross["node"]), cross["dP"], cross["clock"], -1)
            if entry["event"] in ("birth", "credit"):
                row = entry.get("taken_row", rec0)
                for take in entry.get("takes", [entry]):
                    where = tuple(take["node"])
                    first = entry["event"] == "credit" or (where, row, at) not in unbooked
                    if first:
                        unbooked.add((where, row, at))
                        recoil(node, where, take["dP_take"], take["clock"], -1)
                    if "taken_levels" in take:
                        advance(node, take.get("hand", row), where, tuple(take["taken_levels"]))
                        node.strikes[where] += 1
                        continue
                    if not first:
                        continue
                    toward = 1 if take["taker"][1] > take["taker"][0] else -1
                    count = 0 if "laid_back" in take else toward * take["levels"]
                    move(node, row, take["taken_line"], where, -count)
                    node.slack[where] -= int(take["taken"]) - int(take["booked"])
                node.strikes[at] -= 1
        del node.log[frame["log_from"]:]
        node.wound, node.crossed, node.beats = frame["counters"]

    sk.fitted, sk.undo = fitted, undo
    import event_universe.clock as ck
    if hasattr(ck, "undo"):
        ck.undo = undo
