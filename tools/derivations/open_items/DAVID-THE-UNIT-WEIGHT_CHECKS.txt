   """The bang with the held field at a given weight (argv: weight, ticks, out json): the chart with the rewind of 20, saved."""
   import os, sys, json
   sys.set_int_max_str_digits(0)
   WT = os.environ["WT"]
   sys.path.insert(0, WT + "/src")
   from event_universe.run import chart
   weight, ticks, out = int(sys.argv[1]), int(sys.argv[2]), sys.argv[3]
   world = json.load(open(WT + "/tests/worlds/world_bang.json"))
   uni = json.load(open(WT + "/tests/worlds/" + world["universe"]))
   uni = dict(uni, rows=[dict(r, weight=weight) if "weight" in r else r for r in uni["rows"]])
   found = json.loads(json.dumps(chart(uni, dict(world, intervals=ticks), 20, lambda line: None)))
   json.dump(found, open(out, "w"))
