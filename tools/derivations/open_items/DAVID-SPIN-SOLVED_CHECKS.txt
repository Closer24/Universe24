"""A committed record's world with every gapped plane row of its universe named as one row of four lines (a doublet),
run forward n intervals and back b: births by plane pair, clicks by plane, the run back exact."""
import copy, json, sys, collections
sys.set_int_max_str_digits(0)
from event_universe.node import Node
from event_universe.run import state_hash

def doubled(universe):
    u = copy.deepcopy(universe)
    den, gamma = int(u["integers"]["den"]), int(u["integers"]["node_clock"])
    clock = den - den // gamma
    for r in u["rows"]:
        if r.get("pair") and int(r["shape"]) == 2 and int(r["pair"][0]) not in (den, clock):
            r["shape"] = 4
    return u

if __name__ == "__main__":
    record = json.load(open(sys.argv[1]))
    n, back = int(sys.argv[2]), int(sys.argv[3])
    u = doubled(record["universe"]) if (len(sys.argv) < 5 or sys.argv[4] != "plain") else record["universe"]
    node = Node(u, dict(record["world"], intervals=n))
    recs = node.records()
    print("planes:", [(k, r.pair[1] - r.pair[0], None if r.partner is None else recs.index(r.partner)) for k, r in enumerate(recs)])
    kept = []
    for _ in range(n):
        node.step(); kept.append(state_hash(node))
    births = [e for e in node.log if e["event"] == "birth"]
    pairs = collections.Counter((e["line_pair"], e.get("plus_line")) for e in births)
    clicks = collections.Counter(e["line"] for e in node.log if e["event"] == "credit")
    print(f"{n} intervals: births {len(births)} by (- plane, + plane) {dict(pairs)}; clicks by plane {dict(clicks)}; refused_lay {node.counts['refused_lay']}")
    for _ in range(back):
        node.step(-1)
    print(f"the run back {back}: exact {state_hash(node) == kept[n - back - 1]}")

for rec in sorted({k[0] for k in by_charge}):
    dist = {q: n for (r, q), n in sorted(by_charge.items()) if r == rec}
    print(f"  plane {rec}: |W| div e at its clicks {dist}")
print(f"  the census, the most |W| div e of a matter plane at any Node at any interval: {dict(census)}")

== senses_at_clicks.py: the count per sense at the clicks, on the head
========================================================================
"""At every click, the clicking plane's hand at its Node split into its two senses (the plane's two rotating parts at
the family's rest turn: z_now = a + b, z_before = a e^(i w) + b e^(-i w), cos w = num / den), each part's charge in e
(|a|^2 s / den over T div 2): N+ and N-, the count of quanta of each sense Pauli speaks of; the census of the larger
of the two over every matter plane, Node and interval. A reading in floats beside the engine, the engine untouched."""
import json, os, sys, collections, math
sys.set_int_max_str_digits(0)
sys.path.insert(0, sys.argv[0].rsplit("/", 1)[0])
from doublet_world import doubled
from event_universe import bodies as B, close as C
from event_universe.core.paces import sine_den
from event_universe.node import Node

record = json.load(open(sys.argv[1]))
universe = doubled(record["universe"]) if os.environ.get("DOUBLET") == "1" else record["universe"]
world = record["world"]
at_clicks, census = collections.Counter(), collections.Counter()
orig = B.bodies

def counts(row, lines, at, e, den):
    num = row.pair[0]; s = sine_den(num, den)
    w = math.atan2(s, num)
    zn = complex(int(lines[0].now[at]), int(lines[1].now[at])); zb = complex(int(lines[0].before[at]), int(lines[1].before[at]))
    ei = complex(math.cos(w), math.sin(w))
    a = (zb - zn / ei) / (ei - 1 / ei); b = zn - a
    k = (s / den) / e
    return abs(a) ** 2 * k, abs(b) ** 2 * k

def watched(node, plane_starts, owns, born_nodes, frozen, r_emitter):
    e, den = node.A * node.B, node.den
    before = len(node.log)
    orig(node, plane_starts, owns, born_nodes, frozen, r_emitter)
    for ent in node.log[before:]:
        if ent["event"] != "credit":
            continue
        rec, at = ent["line"], tuple(ent["node"])
        n1, n2 = counts(node.records()[rec], plane_starts[rec], at, e, den)
        at_clicks[(rec, int(max(n1, n2) + 0.5), int(min(n1, n2) + 0.5))] += 1
        census[rec] = max(census[rec], round(max(n1, n2), 2))
C.bodies = watched
node = Node(universe, world)
for _ in range(world["intervals"]):
    node.step()
print(f"{sys.argv[1].split('/')[-1]}{' as doublets' if os.environ.get('DOUBLET') == '1' else ''}: {world['intervals']} intervals")
for rec in sorted({k[0] for k in at_clicks}):
    dist = collections.Counter()
    for (r, hi, lo), n in at_clicks.items():
        if r == rec:
            dist[(hi, lo)] += n
    top = sorted(dist.items(), key=lambda kv: -kv[1])[:6]
    print(f"  plane {rec}: (the larger sense, the smaller sense) in quanta, rounded, at its clicks, the most frequent: {top}; the largest at a click {max(h for h, l in dist)}")
print(f"  the largest single sense at any click, per plane: {dict(census)}")
