"""Probe A: the kinds of the six readings under the 48 (the cube's 24 rotations and 24 mirrors).

Integers only (Python ints in numpy object arrays; no float decides anything). Every reading is computed
with the head's own functions where the head has one (regulation.stress_of, regulation.tensor_parts,
hands.lead, meshes.pawls, meshes.gradient, meshes.cross, hands.dot, hands.sweep, unwind.push_of,
symmetries.moment, meshes.train_twists), on random integer fields on an L x L x L torus, and on each of
the field's 48 images (the signed permutation matrices about a node-clock).  For every reading the probe
checks that the image's reading is D(R) applied to the reading, D(R) the claimed representation, at every
site; it then names the kind from D's characters (O's table; g/u by the inversion).

Run: venv314/bin/python -I kinds48.py
"""

def tally(name: str, ok: bool) -> None:

    COUNTS[name] = [runs + 1, fails + (not ok)]


def field(k: int = 60) -> np.ndarray:
    return np.array(

# ---- the 48: R e_a = s_a e_{pi(a)} ----
def det3(m: list[list[int]]) -> int:
    return (
        m[0][0] * (m[1][1] * m[2][2] - m[1][2] * m[2][1])
        - m[0][1] * (m[1][0] * m[2][2] - m[1][2] * m[2][0])
        + m[0][2] * (m[1][0] * m[2][1] - m[1][1] * m[2][0])
    )


GROUP = []
for perm in itertools.permutations(range(3)):
    for signs in itertools.product((1, -1), repeat=3):
        R = [[0] * 3 for _ in range(3)]
        for a in range(3):
            R[perm[a]][a] = signs[a]
        GROUP.append((perm, signs, R, det3(R)))
assert len(GROUP) == 48 and sum(d == 1 for *_, d in GROUP) == 24


def matmul(m: list[list[int]], v: list[int]) -> list[int]:
    return [sum(m[b][a] * v[a] for a in range(3)) for b in range(3)]


def rclass(R: list[list[int]]) -> str:
    """The class of a rotation of O by its trace and shape (E, 8C3, 3C2, 6C4, 6C2')."""
    tr = R[0][0] + R[1][1] + R[2][2]
    if tr == 3:
        return "E"
    if tr == 0:
        return "C3"
    if tr == 1:
        return "C4"
    diagonal = all(R[b][a] == 0 for a in range(3) for b in range(3) if a != b)
    return "C2" if diagonal else "C2'"


def site(R: list[list[int]], x: tuple[int, ...], centre: int) -> tuple[int, ...]:
    return tuple((centre + v) % L for v in matmul(R, [xi - centre for xi in x]))


def image(f: np.ndarray, R: list[list[int]], centre: int = C) -> np.ndarray:
    """The field's image: f'(R x) = f(x) about the node-clock `centre`."""
    out = np.empty_like(f)
    for x in np.ndindex(f.shape):
        out[site(R, x, centre)] = f[x]
    return out


def step(y: tuple[int, ...], b: int, d: int) -> tuple[int, ...]:
    return tuple((v + (d if k == b else 0)) % L for k, v in enumerate(y))

CLASSES = ("E", "C3", "C2", "C4", "C2'")
SIZES = {"E": 1, "C3": 8, "C2": 3, "C4": 6, "C2'": 6}

    "A2": (1, 1, 1, -1, -1),

def kind(rep) -> str:
    """The decomposition of a representation D (a function of R and det R giving a matrix) into O_h's kinds."""
    chi_rot = {c: None for c in CLASSES}
    parity_of_class: dict[str, set[int]] = {c: set() for c in CLASSES}
    for _perm, _signs, R, d in GROUP:
        if d != 1:
            continue
        D = rep(R, 1)
        chi = sum(D[i][i] for i in range(len(D)))
        c = rclass(R)
        assert chi_rot[c] in (None, chi)
        chi_rot[c] = chi
        minus = [[-v for v in row] for row in R]  # the mirror -R, the inversion times R
        Dm = rep(minus, -1)
        chi_m = sum(Dm[i][i] for i in range(len(Dm)))
        parity_of_class[c].add(0 if chi == 0 and chi_m == 0 else (1 if chi_m == chi else -1 if chi_m == -chi else 99))
    parts = []
    for name, row in TABLE.items():
        n24 = sum(SIZES[c] * chi_rot[c] * row[k] for k, c in enumerate(CLASSES))
        assert n24 % 24 == 0
        if n24:
            parts.append(f"{n24 // 24 if n24 // 24 > 1 else ''}{name}")
    pars = set().union(*parity_of_class.values()) - {0}
    g_u = "g (kept by the inversion)" if pars == {1} else "u (reversed by the inversion)" if pars == {-1} else f"mixed {pars}"
    return " + ".join(parts) + f", {g_u}"


def eye_rep(R, d):
    return [[1]]


def vec_rep(R, d):
    return R


def axial_rep(R, d):
    return [[d * v for v in row] for row in R]


def pseudo_rep(R, d):
    return [[d]]


def perm_rep(R, d):
    return [[abs(v) for v in row] for row in R]


def pseudo_perm_rep(R, d):
    return [[d * abs(v) for v in row] for row in R]


def traceless_rep(R, d):
    """The permutation representation on the traceless diagonals, in the basis (e_0 - e_2, e_1 - e_2)."""
    P = perm_rep(R, d)

    def act(v):
        return [sum(P[b][a] * v[a] for a in range(3)) for b in range(3)]

    basis = ([1, 0, -1], [0, 1, -1])
    cols = []
    for v in basis:
        w = act(v)  # w = alpha (1, 0, -1) + beta (0, 1, -1): alpha = w_0, beta = w_1
        cols.append([w[0], w[1]])
    return [[cols[j][i] for j in range(2)] for i in range(2)]


PAIRS = ((1, 2), (2, 0), (0, 1))  # yz, zx, xy


def shear_rep(R, d):
    D = [[0] * 3 for _ in range(3)]
    for k, (a, b) in enumerate(PAIRS):
        ia, sa = next((i, R[i][a]) for i in range(3) if R[i][a])
        ib, sb = next((i, R[i][b]) for i in range(3) if R[i][b])
        j = next(j for j, (p, q) in enumerate(PAIRS) if {p, q} == {ia, ib})
        D[j][k] = sa * sb
    return D


# ---- the readings ----
def bond_momentum(nows: list[np.ndarray], befores: list[np.ndarray], pinion: int) -> list[np.ndarray]:
    """The push's momentum on the bond (x, x + e_a) (unwind.push_of's docstring):
    p_a(x) = 2 pinion SUM_gears [now(x + a) before(x) - now(x) before(x + a)] = 2 pinion [Re conj(z_now(x+a)) z_before(x) - Re conj(z_now(x)) z_before(x+a)]."""
    found = []
    for a in range(AXES):
        up = spoke_of(a, 0)
        terms = [across(n, up) * b - n * across(b, up) for n, b in zip(nows, befores, strict=True)]
        found.append((pinion + pinion) * sum(terms[1:], start=terms[0]))
    return found


def total(arrs: list[np.ndarray]) -> list[int]:
    return [int(sum(a.reshape(-1))) for a in arrs]


def moment_total(bond: list[np.ndarray]) -> list[int]:
    """L = SUM over bonds of arm(x) x (p_a(x) e_a), arm = 2 (x - C) doubled from the centre node-clock
    (symmetries.angular_momentum's arm 2 x - (cube - 1)); only the arm's transverse parts enter, the same at both ends."""
    found = [0, 0, 0]
    for x in np.ndindex(L, L, L):
        arm = [2 * (xi - C) for xi in x]
        for a in range(AXES):
            push = [0, 0, 0]
            push[a] = int(bond[a][x])
            found = [f + m for f, m in zip(found, symmetries.moment(arm, push), strict=True)]
    return found

def board(clock):

def bond_p(nows, befores, pinion):
    found = []
    for a in range(AXES):
        up = spoke_of(a, 0)
        terms = [across(n, up) * b - n * across(b, up) for n, b in zip(nows, befores, strict=True)]
        found.append((pinion + pinion) * sum(terms[1:], start=terms[0]))
    return found


def readings_of(clock, state):
    """(P, L, Q) of a board: the push's momentum, its moment about the great clock's centre (doubled arm), the charge."""
    P, Lm, Q = [0, 0, 0], [0, 0, 0], 0
    for h, gears in zip(clock.hands(), state, strict=True):
        nows, befores = [g[0] for g in gears], [g[1] for g in gears]
        bond = bond_p(nows, befores, h.ratio[0])
        P = [p + int(sum(b.reshape(-1))) for p, b in zip(P, bond, strict=True)]
        for x in np.ndindex(clock.size):
            arm = [2 * v - clock.cube + 1 for v in x]
            for a in range(AXES):
                if int(bond[a][x]):
                    push = [0, 0, 0]
                    push[a] = int(bond[a][x])
                    Lm = [m + d for m, d in zip(Lm, symmetries.moment(arm, push), strict=True)]
        sw = nows[0] * befores[1] - nows[1] * befores[0]
        Q += coupling_of(h) * int(sum(sw.reshape(-1)))
    return P, Lm, Q


def sweep4(n):
    return int(n[0]) * int(n[3]) - int(n[2]) * int(n[1])

    for v in vs:
