   """The bang with its wave seed varied (argv: name, ticks, out json): the seed's p1 transposed (the swap's image), shifted by a column (a translation), cz 3, the mirror 1, or the quanta 4,000,000; the chart with the rewind of 20, saved."""

   WT = os.environ["WT"]
   sys.path.insert(0, WT + "/src")
   from event_universe.run import chart
   name, ticks, out = sys.argv[1], int(sys.argv[2]), sys.argv[3]
   world = json.load(open(WT + "/tests/worlds/world_bang.json"))
   uni = json.load(open(WT + "/tests/worlds/" + world["universe"]))
   seed = dict(world["seed"][0])
   p = seed["p1"]
   if name == "transposed":
       seed["p1"] = [p[4 * j + i] for i in range(4) for j in range(4)]
   elif name == "shifted":
       seed["p1"] = [p[4 * i + (j + 1) % 4] for i in range(4) for j in range(4)]

       seed["quanta"] = 4000000
   found = json.loads(json.dumps(chart(uni, dict(world, intervals=ticks, seed=[seed]), 20, lambda line: None)))
   json.dump(found, open(out, "w"))
   """The merger's push cross term from its own two ends (a harness): each end's live push read on the tick's start board with this merger's earlier ends unwound (a trial copy), not on the great clock other fittings of the striking have moved; so the cross term is the two ends' own, a function of the start and the merger alone."""
   import os, sys, inspect, textwrap
   sys.path.insert(0, os.environ["WT"] + "/src")
   import event_universe.unspring as UN
   import event_universe.striking as SK
   from event_universe.unwind import advance as _advance
   
   def unwind_ends(clock, reads, rec, at=None):
       hand = clock.hands()[rec]
       found = []
       cross = [0] * UN.AXES
       trial = {}
       for read in reads:
           if hand is not clock.unsprung():
               where, tallied, notches, fall, _k, _slack_was, _due, r = read
               if r not in trial:
                   trial[r] = [g.copy() for g in clock.start_gears[r]]
               start = UN.push_of(clock, clock.start_gears[r], where, notches, clock.hands()[r].ratio[0], -1)
               own = UN.push_of(clock, trial[r], where, notches, clock.hands()[r].ratio[0], -1)
               entry, _moved = UN.unspring_end(clock, rec, read)
               moved = [a - b for a, b in zip(own, start, strict=True)]
               for gear, now, before in zip(trial[r], notches[0::UN.HAND], notches[1::UN.HAND], strict=True):
                   gear.now[where] = gear.now[where] - now
                   gear.before[where] = gear.before[where] - before
               cross = [c + m for c, m in zip(cross, moved, strict=True)]
               found.append(entry)
               continue
           found.append(UN.unsprung_end(clock, rec, read))
       if at is not None and len(found) > 1:
           tally = UN.recoil(clock, at, cross, int(clock.start_rate[at]), 1)
           found[-1]["cross"] = {"node": list(at), "dP": cross, **tally}
       return found
   UN.unwind_ends = unwind_ends
