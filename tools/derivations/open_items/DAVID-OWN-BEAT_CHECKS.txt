    def refs(clock):
        if not hasattr(clock, "ref"):
            clock.ref = {}

    def bobs(clock, hand_starts, winds, fitted_clocks, stopped, r_unsprung):
        """The head's chiming with one change: the beat is read on the hand's own axis, the dot of its notch with the notch at which it began to wind (its reference, kept at the first tick of a winding whose notch is not 0, the crossing timer restarted there), in place of the three gear: the beat the ticks between two zero crossings of that dot (CK-S1's reading of a crossing)."""
        ref = refs(clock)
        rec0 = clock.hands().index(clock.unsprung())
        net = sum((train_lead(hand_starts[rec0], spoke) for spoke in range(SPOKES)), start=0)
        for rec, hand in enumerate(clock.hands()):
            if hand is clock.unsprung():
                continue
            wound_now = np.vectorize(lambda v, w=clock.windings[rec]: 1 if int(v) >= w else 0, otypes=[object])(winds[rec])
            three, twelve = hand.gears
            r3, r12 = ref.get(rec, (np.zeros(clock.size, dtype=object), np.zeros(clock.size, dtype=object)))
            unset = np.vectorize(lambda a, b: 1 if int(a) == 0 and int(b) == 0 else 0, otypes=[object])(r3, r12)
            starting = (wound_now == 1) & ((clock.wound[rec] == 0) | (unset == 1))
            r3 = np.where(starting, three.now, r3)
            r12 = np.where(starting, twelve.now, r12)
            ref[rec] = (r3, r12)
            clock.crossed[rec] = np.where(starting, 0, clock.crossed[rec])
            clock.wound[rec] = np.where(wound_now == 1, clock.wound[rec] + 1, 0)
            d_before = r3 * three.before + r12 * twelve.before
            d_now = r3 * three.now + r12 * twelve.now
            crossing = np.vectorize(lambda b, n: 1 if (int(b) * int(n) < 0 or (int(b) == 0) != (int(n) == 0)) else 0, otypes=[object])(d_before, d_now)
            timed = np.vectorize(lambda v: 1 if int(v) > 0 else 0, otypes=[object])(clock.crossed[rec])
            clock.beats[rec] = np.where((crossing == 1) & (timed == 1), clock.crossed[rec], clock.beats[rec])
            clock.crossed[rec] = np.where(crossing == 1, 1, np.where(timed == 1, clock.crossed[rec] + 1, 0))
            chiming = np.vectorize(lambda st, bt: 1 if (int(bt) > 0 and int(st) >= int(bt)) else 0, otypes=[object])(clock.wound[rec], clock.beats[rec])
            for where in zip(*np.nonzero(chiming == 1), strict=True):

    real_note, real_undo, real_state = RUN.rewind_note, SK.undo, RUN.state

    def rewind_note(clock, *args):
        note = real_note(clock, *args)
        note["counters"] = (*note["counters"], {rec: (a.copy(), b.copy()) for rec, (a, b) in refs(clock).items()})
        return note

    def undo(clock, note):
        real_undo(clock, dict(note, counters=note["counters"][:3]))
        clock.ref = note["counters"][3]

    def state(clock):
        return real_state(clock) + [a for rec in sorted(refs(clock)) for a in refs(clock)[rec]]

    SK.undo = undo
