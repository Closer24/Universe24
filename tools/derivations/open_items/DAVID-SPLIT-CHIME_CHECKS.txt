def pairs(clock):

def bobs(clock, hand_starts, winds, fitted_clocks, stopped, r_unsprung):

    net = sum((train_lead(hand_starts[rec0], spoke) for spoke in range(SPOKES)), start=0)
    for u, d in pairs(clock):
        w = clock.windings[u]
        sw = counter(clock, u)
        split = np.vectorize(lambda x, y: 1 if int(x) < w and int(y) < w and int(x) + int(y) >= w else 0, otypes=[object])(winds[u], winds[d])
        sw[...] = np.where(split == 1, sw + 1, 0)
        for where in zip(*np.nonzero(split == 1), strict=True):

            beat = int(clock.beats[u][at]) or int(clock.beats[d][at])
            if beat <= 0 or int(sw[at]) < beat or at in fitted_clocks or stopped[at] or int(net[at]) == 0:
                continue
            sw[at] = 0
            wu, wd = sweep(hand_starts[u], at), sweep(hand_starts[d], at)
            drawn = vernier(clock, at, clock.split_read[0], w)
            if drawn is None or wu + wd == 0:
                continue
            c, room = drawn
            p, o, wp = (u, d, wu) if c * (wu + wd) < room * wu else (d, u, wd)
            was = [[int(g.now[at]), int(g.before[at])] for rec in (u, d) for g in clock.hands()[rec].gears]
            before = window_wind(clock, (u, d), at)
            for g in clock.hands()[p].gears:
                for arr in (g.now, g.before):
                    v = int(arr[at])
                    n = largest_below(v * v * (wu + wd) // wp)
                    arr[at] = n if v >= 0 else -n
            for g in clock.hands()[o].gears:
                g.now[at], g.before[at] = 0, 0

            clock.slack[at] += carried

def rewind_note(clock, *args):
    note = real_note(clock, *args)
    note["counters"] = (*note["counters"], {u: counter(clock, u).copy() for u, _ in pairs(clock)})
    return note


def undo(clock, note):
    splits = [e for e in clock.log[note["log_from"]:] if e.get("event") == "split"]
    real_undo(clock, dict(note, counters=note["counters"][:3]))
    for e in reversed(splits):
        at = tuple(e["node"])
        gears = [g for rec in e["pair"] for g in clock.hands()[rec].gears]
        for g, (n, bf) in zip(gears, e["was"], strict=True):
            g.now[at], g.before[at] = n, bf
        clock.slack[at] -= e["carried"]
    clock.split_wound = {u: x.copy() for u, x in note["counters"][3].items()}


def state(clock):
