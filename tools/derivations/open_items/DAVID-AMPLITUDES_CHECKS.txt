def clock_of(name, cube=3, seed=()):
    u = json.load(open(UNIVERSES[name]))
    if "amplitude" not in u["integers"]:
        u["integers"]["amplitude"] = 288230376000
    return Clock(u, {"shape": [cube] * 3, "seed": list(seed)})


def vacuum_rule(node, row):
    node.start_load, node.start_stretch = readings.load(node), node.train(REGULATOR).gears[0].now
    return readings.gearing(node, row.ratio, node.start_load, factors(node))


def sprung(node):
    return [(k, r) for k, r in enumerate(node.hands()) if r is not node.unsprung()]

            senses = 1 if sense else 2

            def wind(a):
                return setting.wind_at_clock(node, row, setting.even_notches(node, row, a, sense), rule)

(c) One clock at the vacuum: the riding gear's kick and carry, the head's lay against the carried lay
---------------------------------------- leapfrog.py
"""One clock's gear at the vacuum as the engine steps a riding line (tick.impulse and tick.carry): the momentum P = W (a_now - a_before) + fine - mid(W) at the mean wall W = (w div 2)(X_before + X_now) = w Gamma, kicked by X F = Gamma (-D a) (a uniform hand: no mesh pulls), then one carry at W from mid(W); against the rest orbit a cos(k omega), for the head's lay and the carried lay."""
import math
G, Gam = 16, 1000


def run(num, den, now, before, fine, ticks):
    w = 6 * den * Gam * Gam * G * G
    D = 12 * (den - num) * Gam * Gam * G * G
    W = (w // 2) * (Gam + Gam)
    mid = W // 2
    out = [now]
    for _ in range(ticks):
        P = W * (now - before) + fine - mid - Gam * D * now
        v, fine = (P + mid) // W, (P + mid) % W
        before, now = now, now + v
        out.append(now)
    return out


for num, den, a in ((15162854575528411, 15162856759112389, 5690), (2458210083751875, 15162856759112389, 135)):
    w = 6 * den * Gam * Gam * G * G
    W = (w // 2) * (Gam + Gam)
    om = math.acos(num / den)
    head = run(num, den, a, (a * num) // den, W // 2, 1000)

(c) The carried lay as a harness (no engine file changed)
---------------------------------------- carried.py
"""The rest lay carried, as a harness on the head (no engine file changed): each rest lay of the setting (the fourth form and the packets) sets each gear's before and fine wheel, at the clocks it lays, from one carry of the rest velocity: V = w now - (w div rim) N_exact + mid(w), before = now - V div w, fine = V mod w, where now is the lay's own level and N_exact its own numerator (base pinion on the three gear, sense base s on the twelve gear), summed over the senses laid."""
import numpy as np

from common import *
from event_universe.assembly import setting as st
from event_universe.core import acts

_rest_hand = st.rest_hand
CALLS = []
CHANGED = []


def rest_hand(node, row, envelope, w0, amplitude, sense, pairs=None):
    lines = _rest_hand(node, row, envelope, w0, amplitude, sense, pairs)
    if pairs is None:
        num, den = row.ratio[0], node.rim
        base = st.carries(amplitude * envelope, w0)
        CALLS.append((lines, (base * num, sense * base * sine_rim(num, den))))
    return lines


def carry_rest(node, rec, laid_calls):
    """After a rest lay: at the clocks the lay touched, each gear's before and fine from one carry of the lay's rest velocity."""
    row = node.hands()[rec]
    den = node.rim
    for k, g in enumerate(row.gears):
        now = sum(c[0][k].now for c in laid_calls)
        before = sum(c[0][k].before for c in laid_calls)
        exact = sum(c[1][k] for c in laid_calls)
        w = g.teeth
        V = w * now - (w // den) * exact + acts.mid(w)
        touched = (np.asarray(now) != 0) | (np.asarray(exact) != 0)
        carried_before = now - V // w
        shift = np.where(touched, carried_before - before, 0)
        assert all(abs(int(x)) <= 1 for x in shift.reshape(-1)), "the carried before is the head's to one notch"
        assert np.all(np.asarray(g.fine)[touched] == np.asarray(acts.mid(w) + 0 * now)[touched]), "a clock laid twice"
        CHANGED.append(int(sum(1 for x in shift.reshape(-1) if x)))
        g.before = g.before + shift
        g.fine = np.where(touched, V % w, g.fine)


def install():
    st.rest_hand = rest_hand
    real_set_even, real_set_packet = st.set_even, st.set_packet

    def set_even(node, rec, n, sense):
        CALLS.clear()
        real_set_even(node, rec, n, sense)
        carry_rest(node, rec, CALLS[-1:] if sense else CALLS[-2:])

    def set_packet(node, rec, at, h, n, one_sense=None, wave=None, top=1):
        CALLS.clear()
        real_set_packet(node, rec, at, h, n, one_sense, wave, top)
        if wave is not None:
            return
        senses = 1 if one_sense else 2
        carry_rest(node, rec, CALLS[-1 - senses : -1])

    st.set_even, st.set_packet = set_even, set_packet
