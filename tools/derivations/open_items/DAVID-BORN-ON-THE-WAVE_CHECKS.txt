def pair_beat(clock, pair, split):
    """The pair's own beat, updated at this tick; returns its beats array."""
    if not hasattr(clock, "pair_beat"):
        clock.pair_beat = {}
    u, d = pair
    z = lambda: np.zeros(clock.size, dtype=object)
    ref, crossed, beats = clock.pair_beat.setdefault(u, ([z(), z(), z(), z()], z(), z()))
    gears = clock.hands()[u].gears + clock.hands()[d].gears
    unset = np.vectorize(lambda *v: 1 if all(int(x) == 0 for x in v) else 0, otypes=[object])(*ref)
    starting = (split == 1) | ((split > 0) & (unset == 1))
    for i, g in enumerate(gears):
        ref[i][...] = np.where(starting, g.now, ref[i])
    crossed[...] = np.where(starting, 0, crossed)
    was = sum((x * g.before for x, g in zip(ref, gears, strict=True)), start=0)
    read = sum((x * g.now for x, g in zip(ref, gears, strict=True)), start=0)
    crossing = (was * read < 0) | ((was == 0) != (read == 0))
    timed = crossed > 0
    beats[...] = np.where(crossing & timed, crossed, beats)
    crossed[...] = np.where(crossing, 1, np.where(timed, crossed + 1, 0))
    STAGES["split"] += int(np.count_nonzero(split > 0))
    return beats

def up_to(v):
    v = int(v)
    m = -((-abs(v)) // r) * r
    return m if v >= 0 else -m
