"""Item 1: every exact take of the light on a chart (direction 1), the fall fall_of claims at its count against the wind's own fall over the take's window (hands.wind at the clock and its six neighbours, before and after the move), and the closed form with the pull's sign turned."""

ALONE = []


def fall_of(node, at, walls, r, row, direction=1):
    s, claimed, k = real(node, at, walls, r, row, direction)
    if direction == 1 and s:
        line = row.gears[k]
        now, before = int(line.now[at]), int(line.before[at])
        t = 1 if before > now else -1
        window = unwind.window_clocks(node, at)
        was = sum(int(v) for v in hands.wind(node, row.gears, r, window))
        alone_was = int(hands.wind(node, row.gears, r, [at])[0])
        board_was = sum(int(v) for v in hands.wind(node, row.gears, r).reshape(-1))
        line.now[at] = now + t * s
        after = sum(int(v) for v in hands.wind(node, row.gears, r, window))
        alone_after = int(hands.wind(node, row.gears, r, [at])[0])
        board_after = sum(int(v) for v in hands.wind(node, row.gears, r).reshape(-1))
        line.now[at] = now
        ALONE.append((claimed, alone_was - alone_after, board_was - board_after, was - after))
        own, w, reached = int(__import__("numpy").asarray(r.hold)[at]), int(r.teeth), 0
        for port, reach in enumerate(r.grips):
            reached += int(__import__("numpy").asarray(reach)[at]) * int(line.before[unwind.neighbour(node, at, port)])
        pull = own * before + reached
        x, weight = int(node.start_stretch[at]), rule3.wind_weight(node.full_rate, node.G)
        moved = now + t * s
        turned = int(carries(x * (w * (now * now - moved * moved) + t * s * pull), weight))
        rows.append((node.ticks, at, s, walls, claimed, was - after, turned))
    return s, claimed, k

---------------------------------------- fixes.py
"""The close's law: each item's fix as a harness on the head (one engine function replaced, no engine file changed).
1 `wind_fall`: the exact take reads the wind's own fall over its window (hands.wind at the clock and its six neighbours) and its count by the one search by comparisons (rule3.least_at), bounded by the move's distance at 1 and unbounded at -1; the closed form and its price struck.
2 `back_mesh`: the vernier reads each neighbour's grip on the mesh back to the clock (the opposite spoke), R_ji.
3 `lay_back_whole`: the lay-back restores the take's remainder too (the whole take at -1).
5 `one_moment`: the light's take books its push on the gears at the tick's start (CL-S2), as the sprung take does."""
import numpy as np
from common import *


def wind_fall():
    from event_universe import unwind, hands
    from event_universe.core.rule3 import least_at

    def fall_of(node, at, walls, r, row, direction=1):
        gaps = [n * n - b * b for n, b in ((int(z.now[at]), int(z.before[at])) for z in row.gears)]
        k = gaps.index(max(gaps))
        line = row.gears[k]
        now, before = int(line.now[at]), int(line.before[at])
        if walls <= 0 and direction == 1:
            return 0, 0, k
        if now == before and direction == 1:
            return None, 0, k
        t = (1 if before > now else -1) if direction == 1 else (1 if now > before else -1)
        window = unwind.window_clocks(node, at)
        base = sum(int(v) for v in hands.wind(node, row.gears, r, window))

        def fall(s):
            line.now[at] = now + t * s
            found = direction * (base - sum(int(v) for v in hands.wind(node, row.gears, r, window)))
            line.now[at] = now
            return found

        if direction == -1:
            s = least_at(lambda c: fall(c) >= walls)
            return s, fall(s), k
        dist = abs(now - before)
        if fall(dist) < walls:
            return None, fall(dist), k
        s = least_at(lambda c: c >= dist or fall(c) >= walls)
        return s, fall(s), k

    unwind.fall_of = fall_of


def back_mesh():
    from event_universe import draw
    from event_universe.core import acts
    from event_universe.core.rule3 import carries, pitch, remainder
    from event_universe.core.spokes import AXES
    from event_universe.trains import HAND
    from event_universe.unwind import at_clock, window_clocks

    def vernier(node, at, rules, unit):
        modulus = unit * sum(1 for r in node.hands() if r.size == HAND)
        nodes, digits, origin, coarse, w_half = window_clocks(node, at), [], True, True, acts.mid(node.teeth)
        for line, r in rules:
            for k, n in enumerate(nodes):
                wl, key = at_clock(line.teeth, n), (id(r), k, n)
                if key not in node.pitches:
                    node.pitches[key] = (
                        pitch(wl, at_clock(r.spring, n), *[at_clock(x, n) for x in r.grips])
                        if k == 0
                        else pitch(wl, at_clock(r.grips[(k - 1) ^ 1], n))
                    )
                rem, values = int(line.fine[n]), carries(wl, f := node.pitches[key])
                origin = origin and rem in (w_half, acts.flip(w_half, node.teeth, 1))
                coarse = coarse and values <= AXES
                digits.append((carries(rem, f), values + 1))
        if origin or coarse:
            node.counts["refused_origin"] += origin
            return None
        carry, room = 0, 1
        for d, b in sorted(digits):
            carry = remainder(carry * b + d, modulus)
            room = min(room * b, modulus)
        return carry, room

    draw.vernier = vernier
    import event_universe.striking as sk
    sk.vernier = vernier


def lay_back_whole():
    from event_universe import bobs
    import event_universe.unwind as uw
    real_move = bobs.move
    state = {}
    real_chime = bobs.chime

    def chime(node, rec, at, take, frame, start):
        state["take"] = take
        return real_chime(node, rec, at, take, frame, start)

    def move(node, rec0, k, at, count):
        real_move(node, rec0, k, at, count)
        take = state.get("take")
        if take is not None and "remainder_was" in take:
            node.slack[at] = take["remainder_was"]

    bobs.chime, bobs.move = chime, move


def one_moment():
    import event_universe.unspring as un
    real = un.push_of

    def push_of(node, gears, at, notches, pinion, sign=1):
        rec0 = node.hands().index(node.unsprung())
        live = node.hands()[rec0].gears
        if len(gears) == 1 and any(g is gears[0] for g in live):
            k = next(i for i, g in enumerate(live) if g is gears[0])
            return real(node, [node.start_gears[rec0][k]], at, notches, pinion, sign)
        return real(node, gears, at, notches, pinion, sign)

    un.push_of = push_of

"""A committed chart under one fix (argv: the chart's name, the fix): the counts, the turns, the run back and the reading's rows that move, against the committed chart."""
import json, sys, time
from common import *
import fixes
name, fix = sys.argv[1], sys.argv[2]
if fix != "none":
    getattr(fixes, fix)()
from event_universe.run import chart
from event_universe.gates import gates
c = committed(name)
reading = [json.loads(l) for l in open(R + name + ".reading.jsonl") if l.strip()]
back = int((c.get("run_back") or {}).get("intervals", 0))
t = time.time()
found = json.loads(json.dumps(chart(c["universe"], c["world"], back, lambda line: None)))
keys = ("births", "body_clicks", "turns", "taken", "booked", "laid_back", "refused_stop", "refused_lay", "refused_origin")
moved = {k: (c["counts"].get(k), found["counts"].get(k)) for k in keys if c["counts"].get(k) != found["counts"].get(k)}
new_reading = [{k: v} for k, v in json.loads(json.dumps(gates(found))).items()]
rows = [list(r)[0] for r, o in zip(new_reading, reading) if r != o]
verdicts = {}
for r, o in zip(new_reading, reading):
    k = list(r)[0]
    if r != o and isinstance(r[k], dict) and "verdict" in r[k]:
        a, b = str(o[k].get("verdict", ""))[:60], str(r[k].get("verdict", ""))[:60]
        if a != b:
            verdicts[k[:40]] = (a, b)
same = found["nodes"] == c["nodes"] and found["log"] == c["log"]
print(name, fix, "SAME" if same and not moved and not rows else "MOVED", "run back", found["run_back"], "counts moved", moved,
      "reading rows moved", len(rows), "verdicts moved", verdicts, f"{time.time() - t:.0f}s", flush=True)

"""Item 4: every merger (unspringing) of a committed chart: each end's take of the sprung record (its fall against its wall) and each end's light laid at -1 (its rise against the same wall), the excess the close drops, in walls."""
import sys
from common import *
for name in ("chiral", "bodies"):
    log = committed(name)["log"]
    mergers = [e for e in log if e.get("event") == "birth" and "takes" in e]
    fall_x, rise_x, n = [], [], 0
    for e in mergers:
        for t in e["takes"]:
            fall_x.append((t["taken"] - t["booked"]) / t["booked"])
        for t, lay in zip(e["takes"], e.get("exact_lay", [])):
            if lay is not None:
                rise_x.append((lay[2] - t["booked"]) / t["booked"])
        n += 1
    print(f"{name}: {n} unspringings, {len(fall_x)} ends; the sprung take's dropped excess (fall - wall) / wall: "
          f"least {min(fall_x):.3e}, largest {max(fall_x):.3e}, summed {sum(fall_x):.3e} walls; "
          f"the light's lay excess (rise - wall) / wall over {len(rise_x)} ends: least {min(rise_x):.3e}, largest {max(rise_x):.3e}, summed {sum(rise_x):.3e} walls")

3. The vernier's lattices beyond against back
---------------------------------------- vernier_count.py
"""Item 2: at every vernier read of a chart through a tick, each neighbour's mesh lattice as the code reads it (the neighbour's grip on the same spoke, the mesh beyond) against the law's (the grip on the opposite spoke, back to the clock, R_ji)."""
import sys
from common import *
from event_universe import draw
from event_universe.core.rule3 import pitch
from event_universe.unwind import at_clock, window_clocks
from event_universe.clock import Clock
import event_universe.striking as sk
name, n = sys.argv[1], int(sys.argv[2])
real = draw.vernier
tally = {"reads": 0, "lattices": 0, "differ": 0}


def vernier(node, at, rules, unit):
    tally["reads"] += 1
    for line, r in rules:
        for k, m in enumerate(window_clocks(node, at)):
            if k == 0:
                continue
            wl = at_clock(line.teeth, m)
            beyond = pitch(wl, at_clock(r.grips[k - 1], m))
            back = pitch(wl, at_clock(r.grips[(k - 1) ^ 1], m))
            tally["lattices"] += 1
            tally["differ"] += beyond != back
    return real(node, at, rules, unit)


sk.vernier = vernier
c = committed(name)
node = Clock(c["universe"], dict(c["world"], intervals=n))
for _ in range(n):
    node.tick()
print(name, n, "ticks:", tally)

---------------------------------------- orbit_ties.py
"""Item 6: the standard's orbit walked from its seed at a scale X, its half turn by the three tie rules: the table's (readings.standard_angle: the first count whose cosine does not fall), the floor's (features/standard.deviation: the first count whose cosine rises), and the sine's turn (the first count whose sine is not above 0)."""
from common import *
from event_universe.features import standard
for X in (23 * 1000, 1000 * 1000, 288230376000):
    gamma = 1000
    c, s = standard.first_pair(X, gamma)
    cos, sin = [int(c[0])], [int(s[0])]
    for _ in range(4000):
        c, s = standard.act(c, gamma, 1), standard.act(s, gamma, 1)
        cos.append(int(c[0])); sin.append(int(s[0]))
    table = next(n for n in range(1, 4000) if cos[n + 1] >= cos[n])
    floor = next(n for n in range(1, 4000) if cos[n + 1] > cos[n])
    turn = next(n for n in range(1, 4000) if sin[n + 1] <= 0) 
    flat = sum(1 for n in range(1, 40) if cos[n + 1] == cos[n])
    print(f"X {X} ({X // gamma} Gamma): the table's half turn {table}, the floor's {floor}, the sine's turn {turn}; flat steps among the first 40 counts {flat}")

7. The rows' tension range against the tensions alone
---------------------------------------- tensions.py
"""Item 7: the rows' "tension min/max" as run.py reads it (every axis gear of the regulator: tensions, fluxes and shears) against the tensions alone (gears 1 .. AXES), at each tick of a chart."""
import sys
from common import *
from event_universe.clock import Clock
from event_universe.core.spokes import AXES
from event_universe.trains import REGULATOR
name, n = sys.argv[1], int(sys.argv[2])
c = committed(name)
node = Clock(c["universe"], dict(c["world"], intervals=n))
differ = []
for _ in range(n):
    node.tick()
    gears = node.train(REGULATOR).gears
    every = [int(x) for g in gears[1:] for x in g.now.reshape(-1)]
    tensions = [int(x) for g in gears[1:1 + AXES] for x in g.now.reshape(-1)]
    if (min(every), max(every)) != (min(tensions), max(tensions)):
        differ.append((node.ticks, [min(every), max(every)], [min(tensions), max(tensions)]))
print(name, n, "ticks whose row reads another range than the tensions':", len(differ), "of", n, "e.g.", differ[:4])
