               r = rec if twins is None else twins[end]
               notches = unspring.notches_at(clock.hands()[r], where)
               fall = unspring.wind_fall(clock, r, gearing, where, notches, {})
               if fall < tallied:
                   clock.counts["refused_stop"] += cause
                   entry = {"event": "refused", "i": clock.ticks, "node": list(where), "cause": "stop"}
                   clock.log.append(entry | {"fall": fall, "walls": tallied, "owed": tallied, "row": rec})
                   return None
               found.append((where, tallied, notches, fall, None, 0, tallied, r))
               advance(clock, r, where, notches, -1)
               trials.append((r, where, notches))
       finally:
           for r, where, notches in reversed(trials):
               advance(clock, r, where, notches, 1)
           if cache is not None:
               cache.clear()
       return found
   unspring.read_ends = read_ends
   
   _fitted = striking.fitted
   def fitted(clock, at, fits, unwind, rec, c, factors):
       if "takes" in unwind:
           return _fitted(clock, at, fits, unwind, rec, c, factors)
       all_hands = clock.hands()
       involved = sorted({p for f in fits for p in (f[1], hands.twin_of(all_hands, f[1]))})
       window = sorted({w for f in fits for end in (f[2], f[3]) for w in window_clocks(clock, end)})
       def wind():
           return sum(
               int(v)
               for p in involved
               for v in hands.wind(clock, all_hands[p].gears, readings.gearing(clock, all_hands[p].ratio, c, factors), window)
           )
       before = wind()
       out = _fitted(clock, at, fits, unwind, rec, c, factors)
       rise = wind() - before
       clock.slack[at] = clock.slack[at] + int(unwind["booked"]) - rise
       return out
   striking.fitted = fitted
