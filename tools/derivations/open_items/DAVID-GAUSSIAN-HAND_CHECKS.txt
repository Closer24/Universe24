import sys

def per_hand():
    """hands.wind with one division per hand (its two gears' numerators summed) in place of one per gear."""
    from event_universe import hands
    from event_universe.core.rule3 import carries, wind_weight
    from event_universe.core.spokes import SPOKES, across
    from event_universe.trains import zeros
    from event_universe.unwind import neighbour

    def wind(node, lines, r, window=None):
        if window is None:
            at, beyond, x = (lambda a: a), across, node.start_stretch
        else:
            places = [tuple(np.array(w) for w in zip(*window, strict=True))]
            for port in range(SPOKES):
                around = [neighbour(node, where, port) for where in window]
                places.append(tuple(np.array(w) for w in zip(*around, strict=True)))

            def at(a):
                return a[places[0]] if np.ndim(a) else a

            def beyond(a, port):
                return a[places[1 + port]]

            x = node.start_stretch[places[0]]
        full_n, full_b = [line.now for line in lines], [line.before for line in lines]
        nows, befores = [at(n) for n in full_n], [at(b) for b in full_b]
        reached = [at(zeros(node.size)) for _ in lines]
        shared = [at(zeros(node.size)) for _ in lines]
        for port, reach in enumerate(r.grips):
            b_j = [beyond(b, port) for b in full_b]
            n_j = [beyond(n, port) for n in full_n]
            here = at(reach)
            reached = [v + here * bj for v, bj in zip(reached, b_j, strict=True)]
            parts = hands.products([b - bj for b, bj in zip(befores, b_j, strict=True)], [n + nj for n, nj in zip(nows, n_j, strict=True)])
            shared = [s_ + here * part for s_, part in zip(shared, parts, strict=True)]
        own = [
            at(r.teeth) * (nn + bb) - at(r.hold) * nb - nr
            for nn, bb, nb, nr in zip(hands.products(nows, nows), hands.products(befores, befores), hands.products(nows, befores), hands.products(nows, reached), strict=True)
        ]
        wall = wind_weight(node.full_rate, node.G) + wind_weight(node.full_rate, node.G)
        numerators = [(o + o - s_) * x for o, s_ in zip(own, shared, strict=True)]
        found = at(zeros(node.size))
        for k in range(0, len(numerators) - 1, 2):
            found = found + np.asarray(carries(numerators[k] + numerators[k + 1], wall))
        if len(numerators) % 2:
            found = found + np.asarray(carries(numerators[-1], wall))
        return found

    hands.wind = wind


CACHE = {}


PAIRS = {}


def power_pair(X, gamma, n):
    if (X, gamma, n) in PAIRS:
        return PAIRS[(X, gamma, n)]
    PAIRS[(X, gamma, n)] = found = _power_pair(X, gamma, n)
    return found


def _power_pair(X, gamma, n):
    key = (X, gamma)
    if key not in CACHE:
        CACHE[key] = [(1, 0)]
    table = CACHE[key]
    zr, zi = (2 * gamma) * (2 * gamma) - 1, 4 * gamma  # (2 Gamma + i)^2
    while len(table) <= n:
        a, b = table[-1]
        table.append((a * zr - b * zi, a * zi + b * zr))
    a, b = table[n]
    m = (4 * gamma * gamma + 1) ** n
    from event_universe.core.rates import rounded
    return int(rounded(X * a, m)), int(rounded(X * b, m))


def power_circuit():
    from event_universe import readings, hands
    from event_universe.core.rule3 import remainder

    def standard_angle(node, n):
        n = int(n)
        s = 1 if n >= 0 else -1
        n = abs(n)
        if node.circuit_half is None:
            # the half turn: the first count where the cosine stops falling
            k = 1
            while power_pair(node.standard_length, node.full_rate, k + 1)[0] < power_pair(node.standard_length, node.full_rate, k)[0]:
                k += 1
            node.circuit_half = k
        H = node.circuit_half
        if n > H:
            n = remainder(n, H + H)
            if n > H:
                n, s = H + H - n, -s
        c, sn = power_pair(node.standard_length, node.full_rate, n)
        return c, s * sn

    import event_universe.meshes as meshes
    readings.standard_angle = standard_angle
    hands.standard_angle = standard_angle
    meshes.standard_angle = standard_angle

6. Twin hands as one quaternion: the lifts, Kramers and the 4 pi
---------------------------------------- quaternion.py
"""Twin hands as one quaternion q = z_up + z_down j on the four gears (three_up, twelve_up, three_down, twelve_down), all in whole numbers: the Pauli gearing of the twin build (sigma_x swaps the twins, sigma_z turns the down twin over, sigma_y = i sigma_x sigma_z) against the quaternion units, the half turns' lifts U_a = -i sigma_a, Kramers' T, and the 4 pi."""
import itertools

def mul(p, q):
    """Quaternion product of p = (a, b, c, d) = a + b i + c j + d k."""
    a1, b1, c1, d1 = p
    a2, b2, c2, d2 = q
    return (a1*a2 - b1*b2 - c1*c2 - d1*d2, a1*b2 + b1*a2 + c1*d2 - d1*c2, a1*c2 - b1*d2 + c1*a2 + d1*b2, a1*d2 + b1*c2 - c1*b2 + d1*a2)

def as_quaternion(g):
    """(three_up, twelve_up, three_down, twelve_down) -> z_up + z_down j, with z_down j = (x + i y) j = x j + y k."""
    tu, wu, td, wd = g
    return (tu, wu, td, wd)

def gears_of(q):
    return q

# the twin build's gearing on the four gears (DAVID-SPIN-IN-CLOCKS, tick.twin_pulls): up -> (sx, sy, sz)
def sigma(a, g):
    tu, wu, td, wd = g
    up, dn = (tu, wu), (td, wd)
    if a == "x":   # swaps the twins
        nu, nd = dn, up
    elif a == "z":  # turns the down twin over
        nu, nd = up, (-dn[0], -dn[1])
    else:           # sigma_y = i sigma_x sigma_z: up -> (dn[1], -dn[0]) ... as the build spells it
        nu, nd = (dn[1], -dn[0]), (-up[1], up[0])
    return (*nu, *nd)

def times_i(g):
    """i on each twin (the quarter turn of both hands): (three, twelve) -> (-twelve, three)."""
    tu, wu, td, wd = g
    return (-wu, tu, -wd, td)

def U(a, g):
    """The half turn's lift -i sigma_a."""
    s = sigma(a, g)
    return tuple(-x for x in times_i(s))

units = {"i": (0, 1, 0, 0), "j": (0, 0, 1, 0), "k": (0, 0, 0, 1)}
basis = [tuple(int(r == c) for c in range(4)) for r in range(4)]
rng = list(itertools.product(range(-2, 3), repeat=4))
found = {}
for a in "xyz":
    for side in ("left", "right"):
        for name, u in units.items():
            for sign in (1, -1):
                uu = tuple(sign * x for x in u)
                ok = all(U(a, g) == (mul(uu, as_quaternion(g)) if side == "left" else mul(as_quaternion(g), uu)) for g in rng)
                if ok:
                    found[a] = f"{'-' if sign < 0 else ''}{name} on the {side}"
print("the lifts U_a = -i sigma_a as quaternion units:", found)
# Q8 closure and -1 = U_a^2
for a in "xyz":
    assert all(U(a, U(a, g)) == tuple(-x for x in g) for g in rng)
print("U_a U_a = -1 on every gear set in [-2, 2]^4: two half turns read -1 (the 4 pi)")
# Kramers: T = (i sigma_y) K, K the twelve gears turned over (the conjugation of both twins)
def K(g):
    tu, wu, td, wd = g
    return (tu, -wu, td, -wd)
def T(g):
    return times_i(sigma("y", K(g)))
assert all(T(T(g)) == tuple(-x for x in g) for g in rng)
print("Kramers T = (i sigma_y) K: T T = -1 on every gear set; T q = K(q) times, on the right:",
      [("-" if sgn < 0 else "") + name for name, u in units.items() for sgn in (1, -1) if all(T(g) == mul(as_quaternion(K(g)), tuple(sgn * x for x in u)) for g in rng)])
