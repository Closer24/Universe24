The spin one half test's short setting (the twin build in the engine; run from the head's tree)
---------------------------------------- full_turn_short.py
"""The spin one half test's setting, shortened: one mode of the heavy lift's up twin on a cube of side c with a held curl along x and the neutral control, the twins' pulls on against off; the overlap with the free run and S_z at each tick, the ticks where the overlap is least and most, and the seconds (argv: the side, the curl, the ticks)."""
import json, sys, time
sys.set_int_max_str_digits(0)
sys.path.insert(0, sys.argv[0].rsplit("/", 1)[0])
from clk import doubled, hold_curl, neutral, twins_of
side, curl, ticks = int(sys.argv[1]), int(sys.argv[2]), int(sys.argv[3])
calibre = doubled(json.load(open("tests/worlds/universe_toy.json")))
pair = tuple(calibre["rows"][2]["pair"])
neutral(); hold_curl(curl, 0, 0)
import event_universe.tick as T
real = T.Tick.twin_pulls
from event_universe.clock import Clock
t0 = time.time()
piece = {"shape": [side] * 3, "intervals": ticks, "seed": [{"lay": "sea", "row": 2, "hand": 0, "quanta": 1, "sense": 1}]}
on, off = Clock(calibre, piece), Clock(calibre, piece)
up, dn = twins_of(on, pair)
gears = lambda n: [*n.hands()[up].gears, *n.hands()[dn].gears]
rows = []
for t in range(1, ticks + 1):
    T.Tick.twin_pulls = real; on.tick()
    T.Tick.twin_pulls = lambda self: None; off.tick()
    a, b = gears(on), gears(off)
    dot = sum(int((x.now * y.now).sum()) for x, y in zip(a, b)); norm = sum(int((y.now * y.now).sum()) for y in b)
    u = sum(int((x.now * x.now).sum()) for x in a[:2]); d = sum(int((x.now * x.now).sum()) for x in a[2:])
    rows.append((t, (u - d) / (u + d), dot / norm))
least = min(rows, key=lambda r: r[2]); most = max((r for r in rows if r[0] > least[0]), key=lambda r: r[2], default=None)
print(f"side {side} curl {curl} ticks {ticks}: the overlap least {least[2]:+.3f} at tick {least[0]} (S_z {least[1]:+.3f}), "
      f"then most {most[2]:+.3f} at tick {most[0]} (S_z {most[1]:+.3f}); {time.time() - t0:.1f} s")
