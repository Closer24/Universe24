"""The harness on the engine in clock words (67a7633d5 with DAVID-SPIN-IN-CLOCKS_DIFF.txt): a calibre's sprung hands
named as twin hands, and the tests' three controls, each a replacement of one engine function and none an engine act:
the twins' pulls off (Tick.twin_pulls a no-op), a held curl (tick.offset_curl), the offset's turn of a sprung train off
(meshes.coupling_of 0 for the twins: the neutral control)."""
import copy, sys
sys.set_int_max_str_digits(0)
import numpy as np

def doubled(calibre):
    u = copy.deepcopy(calibre)
    den, gamma = int(u["integers"]["den"]), int(u["integers"]["node_clock"])
    regulating = den - den // gamma
    for r in u["rows"]:
        if r.get("pair") and int(r["shape"]) == 2 and int(r["pair"][0]) not in (den, regulating):
            r["shape"] = 4
    return u

def pulls_off():
    import event_universe.tick as T
    T.Tick.twin_pulls = lambda self: None

def hold_curl(cx, cy, cz, grad_z=(0, 0, 0), x_below=None):
    import event_universe.tick as T
    def held(odd):
        size = odd[0].shape
        z = np.zeros(size, dtype=object)
        for k in range(size[2]):
            z[:, :, k] = k - size[2] // 2
        found = [np.zeros(size, dtype=object) + v + g * z for v, g in zip((cx, cy, cz), grad_z)]
        if x_below is not None:
            mask = np.zeros(size, dtype=bool); mask[:x_below] = True
            found = [np.where(mask, f, 0) for f in found]
        return found
    T.offset_curl = held

def neutral():
    import event_universe.meshes as M, event_universe.tick as T
    real = M.coupling_of
    def coupling(r):
        return 0 if getattr(r, "twin", None) is not None else real(r)
    M.coupling_of = T.coupling_of = coupling

def twins_of(node, pair):
    hands = node.hands()
    up = next(k for k, r in enumerate(hands) if r.ratio == pair and r.twin is not None and hands.index(r.twin) > k)
    return up, hands.index(hands[up].twin)

    node.tick(); kept.append(state_hash(node))
book, unsprung = collections.Counter(), []

    if e["event"] != "birth":
        continue

        taken = [t.get("hand", e["taken_row"]) for t in e["takes"]]

    else:
        for h in (e["line_pair"], e.get("plus_hand", e["line_pair"])):
            if hands[h].twin is not None:
                book["fitted up" if h in up else "fitted down"] += 1

    node.tick(-1)
