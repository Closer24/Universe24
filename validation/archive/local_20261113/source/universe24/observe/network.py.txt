"""Passive, explicitly declared positive-length interpretation of network state.

These functions do not schedule arrivals or evolve signals. length/c is the
declared static-edge crossing time, not evidence that a tensor realizes it.
"""
from fractions import Fraction

from universe24.core.network import Link, NetworkSnapshot


def link_length(snapshot: NetworkSnapshot, link: Link, *, field='length') -> int:
    """Interpret current b directly as a positive length in declared length units."""
    value = snapshot.link_value(link, field).b
    if value <= 0:
        raise ValueError('outside the positive-length chart; retained state is unchanged')
    return value


def static_crossing_time(snapshot: NetworkSnapshot, link: Link, *, c=1, field='length') -> Fraction:
    """Exact length/c readout for a frozen edge; c is a unit conversion, not a tick rate."""
    if type(c) not in (int, Fraction) or c <= 0:
        raise ValueError('c must be an exact positive integer or Fraction')
    return Fraction(link_length(snapshot, link, field=field)) / c
