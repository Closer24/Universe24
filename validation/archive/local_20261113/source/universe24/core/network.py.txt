"""Six-neighbor topology and construction-time allocation over existing State.

Node/Link are immutable addresses, not actors. Each link is an independent entity with one state, addressed
at its positive-axis base for array storage only. Both incident nodes read that same retained triple.
Only Engine advances state; this module supplies no default evolution law.
"""
from dataclasses import dataclass
from typing import NamedTuple

import numpy as np

from .engine import State
from .lattice import DIRECTIONS, LINK_AXES, ZERO, lattice
from .tensor import GearType, Tensor


def _index(value, size, name):
    if type(value) is not int or not 0 <= value < size:
        raise ValueError(f'{name} must be an integer in 0..{size - 1}')
    return value


@dataclass(frozen=True, slots=True)
class CubicNetwork:
    """The existing periodic L^3 topology; no lengths or independent state."""

    L: int

    def __post_init__(self):
        if type(self.L) is not int or self.L < 3:
            raise ValueError('L must be an integer >= 3')

    @property
    def node_count(self):
        return self.L ** 3

    @property
    def link_count(self):
        return 3 * self.node_count

    def node(self, index):
        return Node(self, index)

    def link(self, storage_node, axis):
        return Link(self, storage_node, axis)

    def nodes(self):
        return (self.node(n) for n in range(self.node_count))

    def links(self):
        return (self.link(n, axis) for n in range(self.node_count) for axis in range(3))


@dataclass(frozen=True, slots=True)
class Node:
    network: CubicNetwork
    index: int

    def __post_init__(self):
        if not isinstance(self.network, CubicNetwork):
            raise TypeError('expected a CubicNetwork')
        _index(self.index, self.network.node_count, 'node index')

    @property
    def coords(self):
        return lattice(self.network.L).coords(self.index)

    def link(self, direction):
        """Directions: +x,-x,+y,-y,+z,-z. Negative views use the neighbor storage address."""
        _index(direction, 6, 'direction')
        base = self.index if direction % 2 == 0 else lattice(self.network.L).neighbour(self.index, direction)
        return self.network.link(base, direction // 2)

    @property
    def links(self):
        return tuple(self.link(d) for d in range(6))

    def neighbour(self, direction):
        return self.link(direction).other(self)

    def face(self, direction):
        return FacePort(self, direction)

    @property
    def faces(self):
        return tuple(self.face(d) for d in range(6))


@dataclass(frozen=True, slots=True)
class FacePort:
    """A node's oriented interface, not another node or an extra state bank."""

    node: Node
    direction: int

    def __post_init__(self):
        if not isinstance(self.node, Node):
            raise TypeError('expected a Node')
        _index(self.direction, 6, 'direction')

    @property
    def link(self):
        return self.node.link(self.direction)

    @property
    def peer(self):
        return FacePort(self.link.other(self.node), self.direction ^ 1)


@dataclass(frozen=True, slots=True)
class Link:
    """Independent undirected edge, stored at its positive-axis base, including seams."""

    network: CubicNetwork
    storage_node: int
    axis: int

    def __post_init__(self):
        if not isinstance(self.network, CubicNetwork):
            raise TypeError('expected a CubicNetwork')
        _index(self.storage_node, self.network.node_count, 'link storage node')
        _index(self.axis, 3, 'axis')

    @property
    def index(self):
        return 3 * self.storage_node + self.axis

    @property
    def endpoints(self):
        return tuple(self.network.node(n) for n in lattice(self.network.L).link_endpoints(self.storage_node, self.axis))

    def other(self, node):
        if not isinstance(node, Node) or node not in self.endpoints:
            raise ValueError('node is not an endpoint of this link')
        left, right = self.endpoints
        return right if node == left else left


class RetainedValue(NamedTuple):
    a: int
    b: int
    r: int


def _fields(values, name):
    if type(values) not in (tuple, list):
        raise TypeError(f'{name} must be a list or tuple of field names')
    values = tuple(values)
    if any(type(v) is not str or not v.isidentifier() for v in values):
        raise ValueError(f'{name} must contain identifier strings')
    if len(set(values)) != len(values):
        raise ValueError(f'duplicate {name}')
    return values


@dataclass(frozen=True, slots=True)
class NetworkLayout:
    """Each node field costs one channel; each link field costs three.

    Field names have no dynamical meaning. Even a field named length evolves
    only according to the caller's explicitly supplied polynomial tensor.
    """

    node_fields: tuple[str, ...] = ()
    link_fields: tuple[str, ...] = ('length',)

    def __post_init__(self):
        object.__setattr__(self, 'node_fields', _fields(self.node_fields, 'node fields'))
        object.__setattr__(self, 'link_fields', _fields(self.link_fields, 'link fields'))
        if not 1 <= self.channel_count <= 48:
            raise ValueError('layout requires 1..48 total channels per node')

    @property
    def channel_count(self):
        return len(self.node_fields) + 3 * len(self.link_fields)

    @property
    def gears(self):
        return (tuple(GearType(self.node_name(f)) for f in self.node_fields)
                + tuple(GearType(self.link_name(axis, f), link=axis)
                        for axis in range(3) for f in self.link_fields))

    def node_name(self, field):
        if field not in self.node_fields:
            raise ValueError(f'unknown node field {field!r}')
        return f'node.{field}'

    def link_name(self, axis, field):
        _index(axis, 3, 'axis')
        if field not in self.link_fields:
            raise ValueError(f'unknown link field {field!r}')
        return f'link.{LINK_AXES[axis]}.{field}'

    def node_channel(self, field):
        self.node_name(field)
        return self.node_fields.index(field)

    def link_channel(self, axis, field):
        self.link_name(axis, field)
        return len(self.node_fields) + axis * len(self.link_fields) + self.link_fields.index(field)

    def node_read(self, field, direction=None, *, history='b'):
        """Tensor factor relative to the target's storage node."""
        if direction is not None:
            _index(direction, 6, 'direction')
        if history not in ('a', 'b'):
            raise ValueError('tensor factors may read only a or b')
        return self.node_name(field), ZERO if direction is None else DIRECTIONS[direction], history

    def link_read(self, direction, field, *, history='b'):
        """Incident-link tensor factor; negative directions address the neighboring storage slot."""
        _index(direction, 6, 'direction')
        if history not in ('a', 'b'):
            raise ValueError('tensor factors may read only a or b')
        return self.link_name(direction // 2, field), ZERO if direction % 2 == 0 else DIRECTIONS[direction], history

    def tensor(self, monomials):
        """Compile only an explicitly supplied law; always enforce physical support."""
        return Tensor(self.gears, monomials, strict_support=True)

    def manifest(self):
        return [dict(index=i, name=g.name, label=g.label,
                     owner='network.nodes' if g.link is None else f'network.links.{LINK_AXES[g.link]}',
                     purpose='complete generic node field' if g.link is None else 'independent shared link state; positive-axis base is storage only',
                     link=g.link) for i, g in enumerate(self.gears)]

    def uniform_state(self, L, *, nodes, links):
        """Preparation only. Require every complete triple, including all three axes.

        nodes={field: (a,b,r)}; links={field: ((ax,bx,rx),(ay,by,ry),(az,bz,rz))}.
        The returned ordinary State may be further prepared before sealing.
        """
        network = CubicNetwork(L)
        if set(nodes) != set(self.node_fields) or set(links) != set(self.link_fields):
            raise ValueError('provide every declared node/link field, with no extras')
        triples = []
        for field in self.link_fields:
            if type(links[field]) not in (list, tuple) or len(links[field]) != 3:
                raise ValueError('each link field requires three axis triples')
        triples.extend(nodes[field] for field in self.node_fields)
        triples.extend(links[field][axis] for axis in range(3) for field in self.link_fields)
        for triple in triples:
            if type(triple) not in (tuple, list) or len(triple) != 3 or any(type(v) is not int for v in triple):
                raise ValueError('each field requires exact integer (a,b,r)')
            if not 0 <= triple[2] < 6:
                raise ValueError('remainder must be in 0..5')
        return State(L, *[np.array([[triple[k]] * network.node_count for triple in triples], dtype=object)
                          for k in range(3)])

    def bind(self, tensor, state):
        return NetworkSnapshot(self, tensor, state)


@dataclass(frozen=True, slots=True)
class NetworkSnapshot:
    """Read-only addresses into a supplied State; no copied or cached link values.

    The caller still owns State and must not mutate it while observing it.
    Rebind each returned Engine snapshot; an old view keeps observing the old one.
    """

    layout: NetworkLayout
    tensor: Tensor
    _state: State

    def __post_init__(self):
        if not isinstance(self.layout, NetworkLayout) or not isinstance(self.tensor, Tensor) or not isinstance(self._state, State):
            raise TypeError('expected NetworkLayout, Tensor and State')
        if self.tensor.gears != self.layout.gears or self._state.G != self.layout.channel_count:
            raise ValueError('tensor/state do not match this network layout')
        if self.tensor.support_violations():
            raise ValueError('network tensor violates strict face support')
        CubicNetwork(self._state.L)

    @property
    def network(self):
        return CubicNetwork(self._state.L)

    def node_value(self, node, field):
        if not isinstance(node, Node) or node.network != self.network:
            raise ValueError('node belongs to a different network')
        return RetainedValue(*self._state.gear(self.layout.node_channel(field), node.index))

    def link_value(self, link, field):
        if not isinstance(link, Link) or link.network != self.network:
            raise ValueError('link belongs to a different network')
        return RetainedValue(*self._state.gear(self.layout.link_channel(link.axis, field), link.storage_node))
