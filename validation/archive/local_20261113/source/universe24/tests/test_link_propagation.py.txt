"""Independent delay graph, complete inverse and geometric negative controls."""
import random
import unittest

import numpy as np

from universe24.core import CubicNetwork, Engine, State
from validation.link_propagation import (LAYOUT, build_tensor, prepare, inspect,
                                         successor_map, lengths_from_seed,
                                         weighted_countercontrol, flat_grid_metric_audit,
                                         rindler_audit)
from validation.network_smoke import fixture
from validation.system_smoke import scalar_step


class LinkPropagationTests(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.tensor = build_tensor()

    def test_all_lengths_all_axes_and_both_directions(self):
        self.assertEqual(self.tensor.G, 31)
        self.assertEqual(self.tensor.rank, 3)
        self.assertEqual(self.tensor.support_violations(), [])
        for length in (1, 2, 4):
            for axis in range(3):
                for sense in ('right', 'left'):
                    seed = prepare(length=length, axis=axis, sense=sense)
                    result = inspect(self.tensor, seed, length + 1, axis=axis, sense=sense)
                    self.assertEqual(result['first_neighbor_arrival_updates'], length)

    def test_heterogeneous_links_add_delays_both_ways(self):
        for sense, source, expected in (('right', 0, {25: 1, 50: 3, 75: 7}),
                                        ('left', 75, {50: 4, 25: 6, 0: 7})):
            seed = prepare(L=5, mixed=True, sense=sense, source=source)
            result = inspect(self.tensor, seed, 8, sense=sense, source=source)
            for node, tick in expected.items():
                self.assertEqual(result['measured_arrivals'][node], tick)

    def test_multiple_payloads_and_unused_buffers_keep_every_integer(self):
        seed = prepare(length=2)
        rng = random.Random(421)
        indices = [LAYOUT.node_channel(p) for p in LAYOUT.node_fields if p != 'clock']
        indices += [LAYOUT.link_channel(axis, field) for axis in range(3)
                    for field in LAYOUT.link_fields if not field.startswith('select')]
        seed.b = seed.b.astype(object)
        for channel in indices:
            seed.b[channel, :] = [rng.randint(-2**80, 2**80) for _ in range(seed.N)]
        successors = successor_map(seed)
        seed.a = seed.b.ravel()[successors].reshape(seed.b.shape).copy()
        clock = LAYOUT.node_channel('clock')
        seed.a[clock, :] = 0
        current, expected = seed, seed.b.copy()
        engine = Engine(self.tensor)
        for tick in range(1, 6):
            after = engine.step(current)
            target = np.empty_like(expected)
            target.ravel()[successors] = expected.ravel()
            target[clock, :] = (1, 0, -1, 0)[tick % 4]
            np.testing.assert_array_equal(after.b, target)
            self.assertTrue(after.same_as(scalar_step(self.tensor, current)))
            self.assertTrue(engine.inverse_step(after).same_as(current))
            current, expected = after, target
        self.assertTrue(engine.rewind(current, 5).same_as(seed))

    def test_unprepared_states_remain_reversible_but_chart_rejects_them(self):
        rng = random.Random(33)
        arrays = [[[rng.randint(-20, 20) for _ in range(27)] for _ in range(31)] for _ in range(2)]
        carries = [[rng.randrange(6) for _ in range(27)] for _ in range(31)]
        seed = State(3, *arrays, carries)
        with self.assertRaises(ValueError):
            lengths_from_seed(seed)
        engine, current = Engine(self.tensor), seed
        for _ in range(3):
            after = engine.step(current)
            self.assertTrue(after.same_as(scalar_step(self.tensor, current)))
            current = after
        self.assertTrue(engine.rewind(current, 3).same_as(seed))
        invalid = prepare(length=2)
        g = LAYOUT.link_channel(0, 'select4')
        invalid.a[g, 0] = invalid.b[g, 0] = 1
        with self.assertRaises(ValueError):
            lengths_from_seed(invalid)

    def test_both_endpoint_inputs_contribute_to_one_independent_link(self):
        layout, tensor, seed = fixture()
        graph = CubicNetwork(3)
        edge = graph.node(0).link(0)
        left, right = edge.endpoints
        g = layout.link_channel(0, 'length')
        result = []
        for a, b in ((0, 0), (1, 0), (0, 1), (1, 1)):
            state = seed.copy()
            state.a[0, :] = state.b[0, :] = 0
            state.a[0, left.index] = state.b[0, left.index] = a
            state.a[0, right.index] = state.b[0, right.index] = b
            after = Engine(tensor).step(state)
            result.append(int(after.b[g, edge.storage_node]))
            view = layout.bind(tensor, after)
            self.assertEqual(view.link_value(left.link(0), 'length'),
                             view.link_value(right.link(1), 'length'))
        self.assertEqual(result, [1, 2, 2, 1])

    def test_preserve_physical_countercontrols(self):
        weighted = weighted_countercontrol()
        self.assertEqual([r['first_neighbor_arrival_updates'] for r in weighted], [1, 1, 1])
        self.assertEqual([r['length_proportional_delay'] for r in weighted], ['passed', 'failed', 'failed'])
        grid = flat_grid_metric_audit()
        self.assertEqual(grid['euclidean_isotropy_under_this_refinement'], 'failed')
        self.assertEqual(len({r['relative_error'] for r in grid['rows']}), 1)
        rindler = rindler_audit()
        self.assertEqual(rindler['einstein_component_residuals'], ['0'] * 16)
        self.assertEqual(rindler['length_only_identifies_general_observer_geometry'], 'failed')
        self.assertEqual(rindler['proper_clock_rate_ratio_x1_to_x0'], 2)
