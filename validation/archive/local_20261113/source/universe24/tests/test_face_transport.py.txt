"""Native coupled face gates, actual passage and bounded in-flight storage."""
import itertools
import tempfile
import unittest
from pathlib import Path

import numpy as np
from universe24.core import CubicNetwork, Engine, FacePort, State
from universe24.system.bundle import create_bundle
from universe24.system.decoder import default_protocol
from universe24.system.runtime import run_bundle, verify_run
from validation.face_transport import (LAYOUT, BOOLEAN, CLOCK, build_tensor,
                                        permutation, prepare, inspect)
from validation.system_smoke import scalar_step


class FaceTransportTests(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.tensor = build_tensor()

    def test_face_ports_are_shared_interfaces_not_extra_nodes(self):
        for L in (3, 4):
            graph = CubicNetwork(L)
            for node in graph.nodes():
                self.assertEqual(len(node.faces), 6)
                for face in node.faces:
                    self.assertIsInstance(face, FacePort)
                    self.assertEqual(face.peer.peer, face)
                    self.assertEqual(face.link, face.peer.link)
                    self.assertEqual(face.peer.node, node.neighbour(face.direction))
            self.assertEqual(graph.node_count, L**3)
            self.assertEqual(graph.link_count, 3*L**3)
        with self.assertRaises(ValueError):
            CubicNetwork(3).node(0).face(6)

    def test_complete_gate_truth_table_both_directions(self):
        self.assertEqual(self.tensor.G, 25)
        self.assertEqual(self.tensor.rank, 4)
        self.assertEqual(self.tensor.support_violations(), [])
        edge = CubicNetwork(3).node(0).link(0)
        for sense in (0, 1):
            source, end = edge.endpoints if sense == 0 else edge.endpoints[::-1]
            for x, y, q in itertools.product((0, 1), repeat=3):
                seed = prepare(payload=False)
                f, s = (LAYOUT.node_channel(f'{r}{sense}') for r in ('fast', 'slow'))
                phase = LAYOUT.link_channel(0, f'phase{sense}')
                opposite = LAYOUT.link_channel(0, f'phase{1-sense}')
                stage = LAYOUT.link_channel(0, f'stage{sense}')
                seed.b[f, source.index], seed.b[s, source.index] = 1+x, 1+y
                seed.b[phase, edge.storage_node] = 1+q
                seed.a = permutation(seed.b, seed.L, inverse=True)
                after = Engine(self.tensor).step(seed)
                self.assertEqual(int(after.b[f, end.index]), 1+(x if q else y))
                self.assertEqual(int(after.b[stage, edge.storage_node]), 1+(y if q else x))
                self.assertEqual(int(after.b[opposite, edge.storage_node]), 1+(q ^ x ^ y))
                np.testing.assert_array_equal(after.b, permutation(seed.b, seed.L))

    def test_measured_delays_all_directions_and_internal_modes(self):
        for q, d, rail in itertools.product((0, 1), range(6), ('fast', 'slow')):
            result = inspect(self.tensor, prepare(phase=q, direction=d, rail=rail), 8, direction=d)
            expected = 1 if ((rail == 'fast') == bool(q)) else 2
            self.assertEqual(result['first_neighbor_arrival'], expected)
            self.assertGreater(result['phase_changes'], 0)

    def test_dense_domain_and_inflight_exit_independent_of_face_changes(self):
        for L, value in ((3, 7), (3, 24), (4, 42)):
            result = inspect(self.tensor, prepare(L=L, random_seed=value), 32)
            self.assertGreater(result['canonical_delay_increases'], 0)
            self.assertGreater(result['canonical_delay_decreases'], 0)
            self.assertGreater(result['inflight_stage_releases'], 0)
            self.assertEqual(result['no_stalled_inflight_stage'], 'passed')
        # Every slot occupied remains an exactly reversible natural state.
        seed = prepare(payload=False)
        seed.b[list(BOOLEAN)] = 2
        seed.a = permutation(seed.b, seed.L, inverse=True)
        self.assertEqual(inspect(self.tensor, seed, 16)['strict_naturals'], 'passed')

    def test_counter_and_face_phases_do_not_create_payload_from_vacuum(self):
        result = inspect(self.tensor, prepare(payload=False), 64)
        self.assertEqual(result['payload_count'], 0)
        self.assertEqual(result['phase_changes'], 0)
        self.assertEqual(result['samples'][-1]['native_counter'], 66)

    def test_saved_complete_bundle(self):
        seed = prepare(random_seed=2026)
        bundle = create_bundle(self.tensor, seed, channels=LAYOUT.manifest(), protocol=default_protocol(),
                               model={'id': 'face-test', 'version': '1', 'domain': 'prepared Boolean face domain'})
        with tempfile.TemporaryDirectory() as temporary:
            path = Path(temporary)/'run'
            saved = run_bundle(bundle, path, 8)
            self.assertEqual(saved, verify_run(path))
            self.assertEqual(saved['channels_per_node'], 25)

    def test_signed_bigint_outside_chart_retains_exact_inverse(self):
        rng = np.random.default_rng(982)
        a, b = (rng.integers(-8, 9, size=(25, 27)).astype(object) for _ in range(2))
        b[0, 0] = 2**80
        seed = State(3, a, b, rng.integers(0, 6, size=(25, 27)))
        engine, current = Engine(self.tensor), seed
        for _ in range(2):
            after = engine.step(current)
            self.assertTrue(after.same_as(scalar_step(self.tensor, current)))
            self.assertTrue(engine.inverse_step(after).same_as(current))
            current = after
        self.assertTrue(engine.rewind(current, 2).same_as(seed))
        with self.assertRaises(ValueError):
            permutation(seed.b, 3)

    def test_one_scalar_interface_delay_is_insufficient_for_both_modes(self):
        fast = inspect(self.tensor, prepare(phase=1, rail='fast'), 3)
        slow = inspect(self.tensor, prepare(phase=1, rail='slow'), 3)
        self.assertEqual((fast['first_neighbor_arrival'], slow['first_neighbor_arrival']), (1, 2))
