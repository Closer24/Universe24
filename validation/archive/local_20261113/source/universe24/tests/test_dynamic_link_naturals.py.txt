"""Coupled native natural domain, inverse oracle and physical falsifiers."""
import itertools
import tempfile
import unittest
from pathlib import Path

import numpy as np

from universe24.core import CubicNetwork, Engine, State
from universe24.system.bundle import create_bundle
from universe24.system.decoder import default_protocol
from universe24.system.runtime import run_bundle, verify_run
from validation.dynamic_link_naturals import (LAYOUT, build_tensor, prepare,
                                              permutation, inspect,
                                              natural_counterexample,
                                              naive_dynamic_delay_control)
from validation.system_smoke import scalar_step


class DynamicNaturalLinkTests(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.tensor = build_tensor()

    def test_independent_link_truth_table_and_two_endpoint_feedback(self):
        self.assertEqual(self.tensor.G, 10)
        self.assertEqual(self.tensor.rank, 4)
        self.assertEqual(self.tensor.support_violations(), [])
        edge = CubicNetwork(3).node(0).link(0)
        left, right = edge.endpoints
        g = LAYOUT.link_channel(0, 'length')
        for x, y, q in itertools.product((0, 1), repeat=3):
            seed = prepare(payload=False)
            seed.b[LAYOUT.node_channel('port1'), left.index] = 1 + x
            seed.b[LAYOUT.node_channel('port0'), right.index] = 1 + y
            seed.b[g, edge.storage_node] = 1 + q
            seed.a = permutation(seed.b, seed.L, inverse=True)
            after = Engine(self.tensor).step(seed)
            self.assertEqual(int(after.b[g, edge.storage_node]), 1 + (q ^ x ^ y))
            self.assertEqual(int(after.b[LAYOUT.node_channel('port0'), left.index]),
                             1 + (x if q else y))
            self.assertEqual(int(after.b[LAYOUT.node_channel('port1'), right.index]),
                             1 + (y if q else x))
            np.testing.assert_array_equal(after.b, permutation(seed.b, seed.L))
            np.testing.assert_array_equal(permutation(after.b, seed.L, inverse=True), seed.b)
            view = LAYOUT.bind(self.tensor, after)
            self.assertEqual(view.link_value(left.link(0), 'length'),
                             view.link_value(right.link(1), 'length'))

    def test_all_six_directions_keep_naturals_but_length_delay_fails(self):
        for length in (1, 2):
            for direction in range(6):
                result = inspect(self.tensor, prepare(length=length, direction=direction),
                                 25, direction=direction)
                self.assertEqual(result['first_receiver_arrival_update'], 1 if length == 1 else 3)
                self.assertEqual(result['link_state_changes'], 25)
                self.assertEqual(result['recurrence']['period'], 24)

    def test_dense_random_preparations_and_nonzero_carries(self):
        for L, seed_value in ((3, 7), (3, 24), (4, 42)):
            seed = prepare(L=L, random_seed=seed_value)
            self.assertEqual(set(int(v) for v in seed.r.flat), {1, 2, 3, 4, 5})
            result = inspect(self.tensor, seed, 64)
            self.assertGreater(result['link_state_changes'], 0)
            self.assertEqual(result['strict_natural_retained_state'], 'passed')

    def test_arbitrary_positive_histories_are_not_naturally_closed(self):
        result = natural_counterexample(self.tensor)
        self.assertEqual(result['new_b'], -3)
        seed = prepare(payload=False)
        seed.a[LAYOUT.node_channel('port0'), 0] = 5
        self.assertTrue(all(np.all(array >= 1) for array in (seed.a, seed.b, seed.r)))
        engine = Engine(self.tensor)
        self.assertTrue(engine.inverse_step(engine.step(seed)).same_as(seed))

    def test_unprepared_signed_bigint_state_still_rewinds_without_repair(self):
        rng = np.random.default_rng(71)
        a, b = (rng.integers(-7, 8, size=(10, 27)).astype(object) for _ in range(2))
        b[0, 0] = 2**80
        seed = State(3, a, b, rng.integers(0, 6, size=(10, 27)))
        engine, current = Engine(self.tensor), seed
        for _ in range(2):
            after = engine.step(current)
            self.assertTrue(after.same_as(scalar_step(self.tensor, current)))
            self.assertTrue(engine.inverse_step(after).same_as(current))
            current = after
        self.assertTrue(engine.rewind(current, 2).same_as(seed))

    def test_saved_bundle_replays_complete_natural_state(self):
        seed = prepare(random_seed=2026)
        bundle = create_bundle(self.tensor, seed, channels=LAYOUT.manifest(),
                               protocol=default_protocol(), model={'id': 'natural-test', 'version': '1',
                                                                  'domain': 'prepared positive Boolean histories'})
        with tempfile.TemporaryDirectory() as temporary:
            destination = Path(temporary) / 'run'
            saved = run_bundle(bundle, destination, 8)
            self.assertEqual(saved, verify_run(destination))
            self.assertEqual(max(saved['maximum_integer_bits_by_step']), 3)

    def test_direct_frozen_delay_merge_loses_its_prepared_domain(self):
        result = naive_dynamic_delay_control()
        self.assertEqual(result['selector_chart_first_failure'], 1)
        self.assertIsNotNone(result['first_negative_nonclock_value'])
