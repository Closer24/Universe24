"""Generic node/link coupling fixture; no physical signal-speed claim.

The four-channel law is deliberately a mechanical example, not a stable wave
or Einstein model. Lengths influence node evolution; endpoint differences
influence lengths. Only the unchanged Engine advances any retained field.
"""
import argparse
import json
from pathlib import Path

from universe24.core import Engine, NetworkLayout
from universe24.observe.network import link_length, static_crossing_time
from universe24.system.bundle import create_bundle, save_bundle
from universe24.system.decoder import default_protocol
from universe24.system.runtime import run_bundle, verify_run
from validation.system_smoke import scalar_step


def fixture(L=3):
    layout = NetworkLayout(node_fields=('signal',), link_fields=('length',))
    signal = layout.node_name('signal')
    local = layout.node_read('signal')
    terms = [(signal, 6, (local,))]
    for direction in range(6):
        length = layout.link_read(direction, 'length')
        terms.extend(((signal, 3, (length, layout.node_read('signal', direction))),
                      (signal, -3, (length, local))))
    for axis in range(3):
        target = layout.link_name(axis, 'length')
        neighbor = layout.node_read('signal', 2 * axis)
        terms.extend(((target, 6, (layout.link_read(2 * axis, 'length'),)),
                      (target, 3, (local, local)), (target, 3, (neighbor, neighbor)),
                      (target, -6, (local, neighbor))))
    tensor = layout.tensor(terms)
    seed = layout.uniform_state(L, nodes={'signal': (0, 0, 4)},
                                links={'length': ((1, 1, 1), (1, 1, 2), (1, 1, 5))})
    seed.a[0, 0] = seed.b[0, 0] = 1
    return layout, tensor, seed


def run(out, *, L=3, steps=4):
    out = Path(out)
    if out.exists():
        raise ValueError('use a new evidence directory')
    layout, tensor, seed = fixture(L)
    bundle = create_bundle(tensor, seed, channels=layout.manifest(), protocol=default_protocol(),
                           model={'id': 'generic-node-link-mechanical-fixture', 'version': '1',
                                  'domain': 'ownership, strict-local coupling and inverse only; no physical wave or metric acceptance',
                                  'length_readout': 'current b in length units; positive chart only',
                                  'static_crossing_time': 'length/c with c=1; not a signal scheduler'})
    save_bundle(bundle, out / 'input')
    report = run_bundle(out / 'input', out / 'run', steps)
    if report != verify_run(out / 'run'):
        raise AssertionError('saved replay mismatch')
    engine, current = Engine(tensor), seed
    for _ in range(steps):
        expected = scalar_step(tensor, current)
        after = engine.step(current)
        if not after.same_as(expected) or not engine.inverse_step(after).same_as(current):
            raise AssertionError('scalar oracle or complete inverse mismatch')
        current = after
    if not engine.rewind(current, steps).same_as(seed):
        raise AssertionError('complete rewind mismatch')
    snapshot = layout.bind(tensor, current)
    for node in snapshot.network.nodes():
        for direction, edge in enumerate(node.links):
            reverse = node.neighbour(direction).link(direction ^ 1)
            if edge != reverse or snapshot.link_value(edge, 'length') != snapshot.link_value(reverse, 'length'):
                raise AssertionError('endpoint views differ')
    lengths = [link_length(snapshot, edge) for edge in snapshot.network.links()]
    edge = snapshot.network.node(0).link(0)
    evidence = {'role': 'generic network mechanical fixture', 'nodes': snapshot.network.node_count,
                'unique_links': snapshot.network.link_count, 'incident_links_per_node': 6,
                'channels_per_node': tensor.G, 'retained_integers_per_node': 3 * tensor.G,
                'steps': steps, 'scalar_oracle': 'passed', 'rewind': 'passed',
                'shared_endpoint_views': 'passed', 'length_range': [min(lengths), max(lengths)],
                'example_static_crossing_time_c1': str(static_crossing_time(snapshot, edge)),
                'peak_integer_bits': max(report['maximum_integer_bits_by_step']),
                'run_id': report['run_id'], 'physical_propagation_at_c': 'not_tested',
                'physical_metric': 'not_tested'}
    (out / 'network.json').write_text(json.dumps(evidence, indent=2) + '\n')
    return evidence


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--out', type=Path, required=True)
    parser.add_argument('--side', type=int, default=3)
    parser.add_argument('--steps', type=int, default=4)
    args = parser.parse_args()
    print(json.dumps(run(args.out, L=args.side, steps=args.steps), indent=2))


if __name__ == '__main__':
    main()
