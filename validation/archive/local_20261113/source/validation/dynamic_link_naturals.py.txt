"""One coupled natural-state candidate and explicit failure controls.

The candidate is a reversible port scatterer with dynamic independent links,
not a completed variable-delay or Einstein model. P=G S is a local permutation:
S swaps opposite node ports; each shared-link G swaps or holds its endpoint
ports and updates the link from both endpoints. The compiler lifts P and its
inverse into unchanged Rule3. Natural-state closure holds on a declared
complete-history sector, not on arbitrary natural triples.
"""
from __future__ import annotations

import argparse
import hashlib
import json
from pathlib import Path

import numpy as np

from universe24.core import CubicNetwork, Engine, NetworkLayout, Tensor
from universe24.core.lattice import DIRECTIONS, ZERO
from universe24.system.bundle import create_bundle
from universe24.system.decoder import default_protocol
from universe24.system.runtime import run_bundle, verify_run
from validation.system_smoke import scalar_step
from validation.link_propagation import (build_tensor as frozen_tensor,
                                         prepare as frozen_prepare,
                                         flat_grid_metric_audit, rindler_audit)

LAYOUT = NetworkLayout(node_fields=tuple(f'port{d}' for d in range(6)) + ('clock',),
                       link_fields=('length',))


def build_tensor():
    """u=6+3*(P_bits(b-1)+P_bits^-1(b-1)); degree<=3, strict face support.

    b=1+bit. On a=1+P_bits^-1(b-1), Rule3 yields b_next=1+P_bits(b-1).
    No shift, clipping or permutation callback is applied by Engine.
    """
    terms = [(gear.name, 6, ()) for gear in LAYOUT.gears]

    def emit(target, coefficient, factors=()):
        # Expand products of (retained_b - 1) at construction time only.
        for mask in range(1 << len(factors)):
            kept = tuple(f for k, f in enumerate(factors) if mask & (1 << k))
            sign = (-1) ** (len(factors) - len(kept))
            terms.append((target, 3 * coefficient * sign, kept))

    def xor3(target, x, y, q):
        for factor in (x, y, q):
            emit(target, 1, (factor,))
        for left, right in ((x, y), (x, q), (y, q)):
            emit(target, -2, (left, right))
        emit(target, 4, (x, y, q))

    for d in range(6):
        target = LAYOUT.node_name(f'port{d}')
        opposite = LAYOUT.node_read(f'port{d ^ 1}')
        neighbor = LAYOUT.node_read(f'port{d}', d)
        q = LAYOUT.link_read(d, 'length')
        # P=G S: q*own_opposite+(1-q)*neighbor_same_port.
        emit(target, 1, (neighbor,))
        emit(target, 1, (q, opposite))
        emit(target, -1, (q, neighbor))
        # P^-1=S G^-1, with q from the opposite incident link.
        previous_q = LAYOUT.link_read(d ^ 1, 'length')
        previous_neighbor = LAYOUT.node_read(f'port{d}', d ^ 1)
        emit(target, 1, (opposite,))
        emit(target, 1, (previous_q, previous_neighbor))
        emit(target, -1, (previous_q, opposite))
    for axis in range(3):
        d = 2 * axis
        target = LAYOUT.link_name(axis, 'length')
        q = LAYOUT.link_read(d, 'length')
        xor3(target, LAYOUT.node_read(f'port{d ^ 1}'), LAYOUT.node_read(f'port{d}', d), q)
        xor3(target, LAYOUT.node_read(f'port{d}'), LAYOUT.node_read(f'port{d ^ 1}', d), q)
    clock = LAYOUT.node_name('clock')
    emit(clock, 2)
    emit(clock, -2, (LAYOUT.node_read('clock'),))
    return LAYOUT.tensor(terms)


def permutation(values, L, *, inverse=False):
    """Independent scalar Boolean circuit, preparation/verification only."""
    bits = np.array(values, dtype=object) - 1
    if bits.shape != (LAYOUT.channel_count, L**3) or not np.all((bits == 0) | (bits == 1)):
        raise ValueError('the prepared natural chart requires a,b values in {1,2}')
    ports = [LAYOUT.node_channel(f'port{d}') for d in range(6)]
    clock = LAYOUT.node_channel('clock')

    def swap_node_ports(array):
        result = array.copy()
        for d, channel in enumerate(ports):
            result[channel] = array[ports[d ^ 1]]
        result[clock] = 1 - array[clock]
        return result

    incoming = bits if inverse else swap_node_ports(bits)
    outgoing = incoming.copy()
    for edge in CubicNetwork(L).links():
        left, right = (node.index for node in edge.endpoints)
        gx, gy = ports[2 * edge.axis], ports[2 * edge.axis + 1]
        gq = LAYOUT.link_channel(edge.axis, 'length')
        x, y, q = (int(incoming[gx, left]), int(incoming[gy, right]),
                   int(incoming[gq, edge.storage_node]))
        # Forward swaps at q=0; inverse swaps at q=1.
        swap = bool(q) if inverse else not bool(q)
        outgoing[gx, left], outgoing[gy, right] = (y, x) if swap else (x, y)
        outgoing[gq, edge.storage_node] = q ^ x ^ y
    if inverse:
        outgoing = swap_node_ports(outgoing)
    return outgoing + 1


def prepare(L=3, *, length=1, source=0, direction=0, payload=True, random_seed=None):
    if type(length) is not int or length not in (1, 2):
        raise ValueError('declared candidate length is 1 or 2')
    graph = CubicNetwork(L)
    graph.node(source).link(direction)
    nodes = {f'port{d}': (1, 1, 1) for d in range(6)}
    nodes['clock'] = (1, 2, 1)
    links = {'length': ((length, length, 1),) * 3}
    state = LAYOUT.uniform_state(L, nodes=nodes, links=links)
    if random_seed is not None:
        rng = np.random.default_rng(random_seed)
        state.b = rng.integers(1, 3, size=state.b.shape, dtype=np.int64)
    elif payload:
        state.b[LAYOUT.node_channel(f'port{direction ^ 1}'), source] = 2
    state.a = permutation(state.b, L, inverse=True)
    state.r = np.array([[1 + (g + n) % 5 for n in range(state.N)]
                        for g in range(state.G)], dtype=np.int64)
    return state


def inspect(tensor, seed, steps, *, source=0, direction=0):
    """Check every retained triple, the coupled link changes and node arrivals."""
    engine, current, expected = Engine(tensor), seed, seed.b
    rows, seen, arrivals = [], {}, {}
    ports = [LAYOUT.node_channel(f'port{d}') for d in range(6)]
    length_channels = [LAYOUT.link_channel(axis, 'length') for axis in range(3)]
    initial_payloads = int(np.sum(seed.b[ports] - 1))
    receiver = CubicNetwork(seed.L).node(source).neighbour(direction).index
    changed_links = 0
    recurrence = None
    for tick in range(steps + 1):
        if not np.array_equal(current.b, expected):
            raise AssertionError('native state differs from independent reversible circuit')
        if not np.array_equal(current.a, permutation(current.b, seed.L, inverse=True)):
            raise AssertionError('complete prepared-history invariant lost')
        if any(not np.all(array >= 1) for array in (current.a, current.b, current.r)):
            raise AssertionError('strict natural-state chart lost')
        if not np.array_equal(current.r, seed.r):
            raise AssertionError('nonzero carries lost')
        if int(np.sum(current.b[ports] - 1)) != initial_payloads:
            raise AssertionError('Boolean payload count changed')
        counts = np.sum(current.b[ports] - 1, axis=0)
        for n in np.flatnonzero(counts):
            arrivals.setdefault(int(n), tick)
        key = tuple(tuple(int(v) for v in array.flat)
                    for array in (current.a, current.b, current.r))
        if key in seen and recurrence is None:
            recurrence = {'first_tick': seen[key], 'repeat_tick': tick, 'period': tick - seen[key]}
        seen.setdefault(key, tick)
        edge = CubicNetwork(seed.L).node(source).link(direction)
        length = int(current.b[LAYOUT.link_channel(edge.axis, 'length'), edge.storage_node])
        rows.append({'tick': tick, 'source_link_length': length,
                     'receiver_payload_count': int(counts[receiver]),
                     'changed_lengths_from_seed': int(np.sum(current.b[length_channels] != seed.b[length_channels]))})
        if tick == steps:
            break
        after = engine.step(current)
        if not after.same_as(scalar_step(tensor, current)) or not engine.inverse_step(after).same_as(current):
            raise AssertionError('scalar oracle or inverse mismatch')
        changed_links += int(np.sum(after.b[length_channels] != current.b[length_channels]))
        expected = permutation(current.b, seed.L)
        current = after
    if not engine.rewind(current, steps).same_as(seed):
        raise AssertionError('complete rewind failure')
    return {'native_updates': steps, 'strict_natural_retained_state': 'passed',
            'a_b_values': [1, 2], 'r_values': [1, 2, 3, 4, 5],
            'independent_scalar_and_circuit_oracles': 'passed', 'complete_rewind': 'passed',
            'initial_payload_count': initial_payloads, 'payload_count_conserved': 'passed',
            'link_state_changes': changed_links, 'first_receiver_arrival_update': arrivals.get(receiver),
            'recurrence': recurrence, 'samples': rows,
            'clock_scope': 'autonomous period-two reference, not operational proper time',
            'scope': 'coupled port scatterer; interpreting the link value as physical length is a hypothesis'}


def naive_dynamic_delay_control():
    """Directly coupling the earlier frozen selectors does not preserve its proof."""
    old = frozen_tensor()
    terms = [(m.target, m.coeff, m.factors) for m in old.monomials]
    # Separate attempted law: endpoint intensity contributes to select2.
    # Do not overwrite the pinned frozen-delay experiment or switch laws mid-run.
    for axis in range(3):
        target = old.gear_index(f'link.{"xyz"[axis]}.select2')
        terms.append((target, 3, [(f'node.right{axis}', ZERO, 'b')]))
        terms.append((target, 3, [(f'node.right{axis}', DIRECTIONS[2 * axis], 'b')]))
    tensor = Tensor(old.gears, terms, strict_support=True)
    current = frozen_prepare(length=2)
    from validation.link_propagation import lengths_from_seed
    first_invalid = None
    first_negative = None
    first_nonclock_negative = None
    nonclock = [g for g in range(tensor.G) if tensor.gears[g].name != 'node.clock']
    for tick in range(7):
        if first_negative is None and any(np.any(array < 0) for array in (current.a, current.b, current.r)):
            first_negative = tick
        if first_nonclock_negative is None and any(np.any(array[nonclock] < 0)
                                                   for array in (current.a, current.b, current.r)):
            first_nonclock_negative = tick
        try:
            lengths_from_seed(current)
        except ValueError:
            if first_invalid is None:
                first_invalid = tick
        current = Engine(tensor).step(current)
    return {'selector_chart_first_failure': first_invalid,
            'first_negative_retained_value': first_negative,
            'first_negative_nonclock_value': first_nonclock_negative,
            'clock_already_has_signed_values': True,
            'scope': 'failed direct merge of the frozen-delay law with endpoint-driven selectors; not an impossibility theorem'}


def natural_counterexample(tensor):
    seed = prepare(payload=False)
    seed.a[LAYOUT.node_channel('port0'), 0] = 5
    after = Engine(tensor).step(seed)
    value = int(after.b[LAYOUT.node_channel('port0'), 0])
    if value != -3:
        raise AssertionError('natural-domain counterexample changed')
    return {'initial_all_retained_values_positive': True,
            'changed_old_a': 5, 'new_b': value,
            'all_natural_initial_states_closed': 'failed',
            'reason': 'a complete positive seed outside the compatible-history sector can leave the naturals; no clipping is applied'}


def source_hashes():
    root = Path(__file__).resolve().parents[1]
    names = ('validation/dynamic_link_naturals.py', 'validation/link_propagation.py',
             'validation/network_smoke.py', 'validation/system_smoke.py',
             'universe24/core/engine.py', 'universe24/core/tensor.py',
             'universe24/core/network.py', 'universe24/core/lattice.py',
             'universe24/system/bundle.py', 'universe24/system/runtime.py',
             'universe24/system/decoder.py')
    return {name: hashlib.sha256((root / name).read_bytes()).hexdigest() for name in names}


def run(out):
    out = Path(out)
    if out.exists():
        raise ValueError('use a new output directory')
    out.mkdir(parents=True)
    tensor = build_tensor()
    sources = source_hashes()
    cases = []

    def execute(name, seed, steps, direction=0):
        bundle = create_bundle(tensor, seed, channels=LAYOUT.manifest(), protocol=default_protocol(),
                               model={'id': 'dynamic-natural-link-scatterer', 'version': '1',
                                      'domain': 'prepared positive Boolean encoding, a=P^-1 b, all nonzero carries; no Einstein acceptance',
                                      'natural_encoding': 'a,b in {1,2}, r in {1,2,3,4,5}; coefficients remain signed integers',
                                      'length_hypothesis': 'current link b in {1,2}; test whether this is a physical length'})
        saved = run_bundle(bundle, out / name, steps)
        if saved != verify_run(out / name):
            raise AssertionError('saved replay differs')
        result = inspect(tensor, seed, steps, direction=direction)
        result.update(name=name, run_id=saved['run_id'], bundle_id=saved['bundle_id'],
                      peak_integer_bits=max(saved['maximum_integer_bits_by_step']),
                      tensor_hash=tensor.fingerprint)
        cases.append(result)

    execute('empty', prepare(payload=False), 32)
    for length in (1, 2):
        for direction in range(6):
            execute(f'length{length}-direction{direction}', prepare(length=length, direction=direction), 48, direction)
    for random_seed in (7, 24, 2026):
        execute(f'random{random_seed}', prepare(random_seed=random_seed), 128)
    execute('L4-long', prepare(L=4, random_seed=42), 512)
    measured_delay = [{'name': case['name'], 'proposed_initial_length': length,
                       'measured_first_arrival': case['first_receiver_arrival_update']}
                      for length in (1, 2) for case in cases
                      if case['name'].startswith(f'length{length}-')]
    if not any(row['proposed_initial_length'] != row['measured_first_arrival']
               for row in measured_delay):
        raise AssertionError('declared length/delay negative control was not reproduced')
    if not all(case['link_state_changes'] > 0 for case in cases if case['initial_payload_count']):
        raise AssertionError('coupled cases must actually update independent links')
    result = {'schema': 'universe24/dynamic-link-naturals/1', 'tensor_hash': tensor.fingerprint,
              'channels_per_node': tensor.G, 'retained_integers_per_node': 3 * tensor.G,
              'maximum_degree': tensor.rank - 1, 'strict_support_violations': len(tensor.support_violations()),
              'cases': cases, 'length_delay_control': measured_delay,
              'total_native_updates': sum(case['native_updates'] for case in cases),
              'naive_merge': naive_dynamic_delay_control(),
              'natural_domain_negative': natural_counterexample(tensor),
              'spatial_geometry_control': flat_grid_metric_audit(), 'einstein_clock_control': rindler_audit(),
              'source_hashes': sources,
              'gates': {'dynamic_shared_links': 'passed', 'strict_naturals_on_prepared_domain': 'passed',
                        'local_degree_budget_and_inverse': 'passed',
                        'unrestricted_natural_state_closure': 'failed',
                        'general_length_equals_transit_time': 'failed',
                        'isotropic_shortest_edge_geometry': 'failed',
                        'proper_clock_identification': 'not_tested',
                        'einstein_regge_field_equations': 'not_tested',
                        'close_to_general_einstein_spacetime': 'blocked'},
              'scope': 'a complete mechanical coupled candidate was run; a full accepted physical model has not been constructed'}
    if sources != source_hashes():
        raise AssertionError('sources changed during experiment')
    (out / 'results.json').write_text(json.dumps(result, indent=2) + '\n')
    return result


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--out', type=Path, required=True)
    args = parser.parse_args()
    result = run(args.out)
    print(json.dumps({k: result[k] for k in ('channels_per_node', 'maximum_degree', 'gates')}, indent=2))


if __name__ == '__main__':
    main()
